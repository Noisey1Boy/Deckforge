   164→    const file = files?.[0];
   165→    if (!file) return;
   166→    if (!/^image\/(png|jpe?g|webp)$/.test(file.type)) {
   167→      toast.error('Use a PNG, JPG or WebP screenshot of your chart.');
   168→      return;
   169→    }
   170→    if (file.size > 6_000_000) {
   171→      toast.error('Image must be under 6 MB.');
   172→      return;
   173→    }

   175→    try {
   176→      const dataUrl = await new Promise<string>((resolve, reject) => {
   177→        const r = new FileReader();
   178→        r.onload = () => resolve(String(r.result));
   179→        r.onerror = reject;
   180→        r.readAsDataURL(file);
   181→      });

   183→        method: 'POST',
   184→        headers: { 'Content-Type': 'application/json' },
   185→        body: JSON.stringify({ image: dataUrl, language }),
   186→      });
   187→      const json = await res.json();
   188→      if (!res.ok) throw new Error(json.error || 'Extraction failed');
   189→      setChartSpecs((prev) => [...prev, json.chart as ChartSpec]);
   190→      toast.success('Chart extracted from image.');
   191→    } catch (e) {
   192→      toast.error(e instanceof Error ? e.message : 'Image analysis failed');

   194→      setBusy(null);

   280→          chartAssignments[String(i)] = pool.shift()!;

   284→
   285→      setDataAssets({
   286→        tables,
   287→        charts,
   288→        diagram: remainingDiagram ? { ...remainingDiagram } : undefined,
   289→        dataContext,
   290→        chartAssignments,
   291→      });

   293→      setView('outline');
   294→    } catch (e) {
   295→      toast.error(e instanceof Error ? e.message : 'Generation failed');
