   151→    if (!files?.length) return;
   152→    setDataError(null);
   153→    setBusy('Reading data…');
   154→    const newTables: ParsedTable[] = [...tables];
   155→    for (const file of Array.from(files).slice(0, 3)) {
   156→      const { tables: ts, error } = await parseFileToTables(file);
   157→      if (error) {
   158→        setDataError(error);
   159→        toast.error(error);
   160→      }
   161→      newTables.push(...ts);
   162→    }
   163→    setTables(newTables.slice(0, 3));
   164→    setChartSpecs([]); // re-plan after new data
   165→    setBusy(null);
   166→  };

   168→  const handleChartImage = async (files: FileList | null) => {
   169→    const file = files?.[0];
   170→    if (!file) return;
   171→    if (!/^image\/(png|jpe?g|webp)$/.test(file.type)) {
   172→      toast.error('Use a PNG, JPG or WebP screenshot of your chart.');
   173→      return;
   174→    }
   175→    if (file.size > 6_000_000) {
   176→      toast.error('Image must be under 6 MB.');
   177→      return;
   178→    }
   179→    setBusy('Reading chart from image…');
   180→    try {
   181→      const dataUrl = await new Promise<string>((resolve, reject) => {
   182→        const r = new FileReader();
   183→        r.onload = () => resolve(String(r.result));
   184→        r.onerror = reject;
   185→        r.readAsDataURL(file);
   186→      });
   187→      const res = await fetch('/api/data/chart-image', {
   188→        method: 'POST',
   189→        headers: { 'Content-Type': 'application/json' },
   190→        body: JSON.stringify({ image: dataUrl, language }),
   191→      });
   192→      const json = await res.json();
   193→      if (!res.ok) throw new Error(json.error || 'Extraction failed');
   194→      setChartSpecs((prev) => [...prev, json.chart as ChartSpec]);
   195→      toast.success('Chart extracted from image.');
   196→    } catch (e) {
   197→      toast.error(e instanceof Error ? e.message : 'Image analysis failed');
   198→    } finally {
   199→      setBusy(null);
   200→      if (imgRef.current) imgRef.current.value = '';
   201→    }

   212→        method: 'POST',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({ mermaid, language }),
   215→      });
   216→      const json = await res.json();
   217→      if (!res.ok) throw new Error(json.error || 'Conversion failed');
   218→      if (json.diagram) {
   219→        setDiagram(json.diagram as MermaidPlan);
   220→        toast.success('Diagram ready to include.');
   221→      } else throw new Error('Could not parse the diagram');
   222→    } catch (e) {
   223→      toast.error(e instanceof Error ? e.message : 'Conversion failed');
   224→    } finally {
   225→      setBusy(null);
   226→    }
   227→  };

   244→      if (wantsAnalysis) {
   245→        setBusy('Analyzing your data…');

   247→          method: 'POST',
   248→          headers: { 'Content-Type': 'application/json' },
   249→          body: JSON.stringify({ tables: tables.map(tableToMeta), language }),
   250→        });
   251→        const json = await res.json();
   252→        if (!res.ok) throw new Error(json.error || 'Analysis failed');
   253→        const plans = (json.charts ?? []) as ChartPlan[];

   257→            return t ? buildChartSpec(t, p) : null;
   258→          })
   259→          .filter((c): c is ChartSpec => !!c);
   260→        charts = [...charts, ...specs];
   261→        dataContext = json.dataContext;
   262→      }
   263→      if (diagramPlan) {
   264→        dataContext = `${dataContext ?? ''}\nA Mermaid flow diagram "${diagramPlan.title}" with ${diagramPlan.nodes.length} steps is available for a "diagram" slide.`;
   265→      }
   266→
   267→      setBusy('Writing the outline…');
   268→      const res = await fetch('/api/generate/outline', {
   269→        method: 'POST',
   270→        headers: { 'Content-Type': 'application/json' },
   271→        body: JSON.stringify({
   272→          prompt, audience, tone, slideCount, language, themeChoice, dataContext,
   273→        }),
   274→      });
   275→      const json = await res.json();
   276→      if (!res.ok) throw new Error(json.error || 'Outline failed');
   277→      const outline = json.outline as DeckOutline;
   278→
   279→      // attach data: first N chart slides in the outline get real chart specs
   280→      const chartAssignments: Record<string, ChartSpec> = {};
   281→      const pool = [...charts];
   282→      let remainingDiagram = diagramPlan;
   283→      outline.slides.forEach((s, i) => {
   284→        if (s.layout === 'chart' && pool.length) {
   285→          chartAssignments[String(i)] = pool.shift()!;
   286→        }
   287→      });
   288→      // if diagram exists but no diagram slide planned, convert the last chart-less content slide? Keep: AI was told about it.
   289→
   290→      setDataAssets({
   291→        tables,
   292→        charts,
   293→        diagram: remainingDiagram ? { ...remainingDiagram } : undefined,
   294→        dataContext,
   295→        chartAssignments,
   296→      });
   297→      setOutline(outline);
   298→      setView('outline');
   299→    } catch (e) {
   300→      toast.error(e instanceof Error ? e.message : 'Generation failed');
   301→    } finally {
   302→      setBusy(null);
   303→    }

   318→        title: parsed.title,

   329→      const pool = [...chartSpecs];

   336→        tables, charts: chartSpecs,
   337→        diagram: diagram ?? undefined,
   338→        dataContext: undefined,
   339→        chartAssignments,
   340→      });
   341→      setOutline(outline);
   342→      setView('outline');
   343→      toast.success(`Outline built — ${outline.slides.length} slides from your markdown.`);

   345→      toast.error(e instanceof Error ? e.message : 'Could not parse markdown');
