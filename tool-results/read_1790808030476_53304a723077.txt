    36→function isTypingTarget(t: EventTarget | null): boolean {
    37→  const el = t as HTMLElement | null;
    38→  if (!el) return false;
    39→  const tag = el.tagName;
    40→  return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || el.isContentEditable;
    41→}

    43→function SlideCard({
