   236→/**
   237→ * Detects the framework from the template HTML (mirrors deck-preview logic).
   238→ */
   239→export function detectFramework(html: string): Framework {
   240→  if (/class="reveal"|Reveal\.initialize|\/revealjs\//i.test(html)) return 'reveal.js'
   241→  if (/id="impress"|impress\(\)\.init|\/frameworks\/impressjs\//i.test(html)) return 'impress.js'
   242→  if (/class="shower"|\/frameworks\/shower\//i.test(html)) return 'shower'
   243→  if (/aria-selected="true"|\/frameworks\/dzslides\//i.test(html) && /<section/i.test(html) && !/class="deck"/i.test(html)) return 'dzslides'
   244→  return 'open-design'
   245→}
   246→
   247→/**
   248→ * Extracts just the design-relevant parts of the template HTML.
   249→ * Returns <head> (fonts, tokens, base CSS) + first 2 slides (for class vocabulary).
   250→ */
   251→function extractTemplateDigest(html: string, css: string | undefined, framework: Framework): string {
   252→  const parts: string[] = []
   253→
   254→  // <head>
   255→  const headMatch = html.match(/<head[^>]*>([\s\S]*?)<\/head>/i)
   256→  if (headMatch) {
   257→    parts.push('=== TEMPLATE <head> (fonts, tokens, base CSS) ===')
   258→    parts.push(headMatch[0])
   259→  }
   260→
   261→  // First 2 slide structures (for class vocabulary)
   262→  let slideMatches: string[] | null = null
   263→  switch (framework) {
   264→    case 'reveal.js': {
   265→      const slidesBlock = html.match(/<div class="slides">([\s\S]*?)<\/div>\s*<\/div>/i)
   266→      if (slidesBlock) slideMatches = slidesBlock[1].match(/<section[^>]*>[\s\S]*?<\/section>/gi)
   267→      break
   268→    }
   269→    case 'impress.js': {
   270→      const impressBlock = html.match(/<div id="impress">([\s\S]*?)<\/div>\s*(?:<[^>]+>\s*)*<(?:script|\/body)/i)
   271→      if (impressBlock) slideMatches = impressBlock[1].match(/<div[^>]*class="[^"]*step[^"]*"[^>]*>[\s\S]*?<\/div>/gi)
   272→      break
   273→    }
   274→    case 'shower':
   275→      slideMatches = html.match(/<section[^>]*class="[^"]*slide[^"]*"[^>]*>[\s\S]*?<\/section>/gi)
   276→      break
   277→    case 'dzslides':
   278→      slideMatches = html.match(/<section[^>]*>[\s\S]*?<\/section>/gi)
   279→      break
   280→    default:
   281→      slideMatches = html.match(/<section[^>]*class="[^"]*slide[^"]*"[^>]*>[\s\S]*?<\/section>/gi)
   282→  }
   283→  if (slideMatches && slideMatches.length > 0) {
   284→    parts.push('\n=== TEMPLATE first slides (class vocabulary reference; copy class names) ===')
   285→    parts.push(slideMatches.slice(0, 2).join('\n'))
   286→  }
   287→
   288→  // Runtime scripts (for framework-based templates)
   289→  if (framework === 'reveal.js' || framework === 'impress.js' || framework === 'shower') {
   290→    const allScripts = html.match(/<script[^>]*>[\s\S]*?<\/script>/gi)
   291→    if (allScripts && allScripts.length > 0) {
   292→      parts.push(`\n=== ${framework} script tags (for context — DO NOT include in slide output) ===`)
   293→      parts.push(allScripts.join('\n'))
   294→    }
   295→  }
   296→
   297→  if (css) {
   298→    parts.push('\n=== TEMPLATE style.css ===')
   299→    parts.push(css)
   300→  }
   301→
   302→  return parts.join('\n')
   303→}

   305→/**
   306→ * Builds the system prompt for a single slide.
   307→ * Much smaller than the full-deck prompt — focuses on ONE slide's layout + content.
   308→ */
   309→export function buildSlideSystemPrompt(params: SlidePromptParams): string {
   310→  const aud = params.audience?.trim() || 'general business audience'
   311→  const tn = params.tone?.trim() || 'professional, confident, clear'
   312→  const vb = params.verbosity?.trim() || 'standard'
   313→  const digest = extractTemplateDigest(params.templateHtml, params.templateCss, params.framework)
   314→  const layoutHint = LAYOUT_GUIDANCE['title'] ?? 'Use the template class vocabulary.'
   315→
   316→  // Per-framework slide-element contract
   317→  let slideElementContract = ''
   318→  let notesTag = '<div class="notes">'
   319→  switch (params.framework) {
   320→    case 'reveal.js':
   321→      slideElementContract = `Output a single <section>...</section> element (NO class="slide", NO wrapping <div class="slides">). reveal.js manages visibility.`
   322→      notesTag = '<aside class="notes">'
   323→      break
   324→    case 'impress.js':
   325→      slideElementContract = `Output a single <div class="step slide" data-x="..." data-y="..." [data-rotate="..."] [data-scale="..."]>...</div> element. Position it on the 3D canvas — vary data-x by ${1400 * params.slideIndex} from the previous slide, optionally rotate or scale.`
   326→      notesTag = '<div class="notes">'
   327→      break
   328→    case 'shower':
   329→      slideElementContract = `Output a single <section class="slide">...</section> element. Shower manages visibility.`
   330→      notesTag = '<details class="notes">'
   331→      break
   332→    case 'dzslides':
   333→      slideElementContract = `Output a single <section>...</section> element. Use aria-selected="true" only if this is slide 0.`
   334→      notesTag = '<div role="note">'
   335→      break
   336→    default:
   337→      slideElementContract = `Output a single <section class="slide" data-title="...">...</section> element. Use the template's class vocabulary: .kicker, .h1/.h2/.h3, .lede, .card, .pill, .grid.g2/.g3/.g4, .stack, .center, .tc, .big-number, .big-label, .quote, .timeline-item, .delta.up/.down.`
   338→      notesTag = '<div class="notes">'
   339→  }
   340→
   341→  return `You are an expert presentation designer. Generate the HTML for ONE slide.
   342→
   343→# TASK
   344→Generate exactly one slide element for a ${params.framework} deck titled "${params.deckTitle}".
   345→This is slide ${params.slideIndex + 1} of ${params.slideTotal}.
   346→The slide spec (title, layout, content markdown, speaker notes) arrives in the user message.
   347→
   348→# CONTEXT
   349→- Audience: ${aud}
   350→- Tone: ${tn}
   351→- Verbosity: ${vb}
   352→
   353→# FRAMEWORK CONTRACT (${params.framework})
   354→${slideElementContract}
   355→Speaker notes go inside ${notesTag}...</${notesTag.slice(1, -1).split(' ')[0]}> within the slide element.
   356→
   357→# TEMPLATE DESIGN DIGEST
   358→${digest}
   359→
   360→# LAYOUT HINT
   361→${layoutHint}
   362→
   363→# QUALITY FLOORS
   364→${FLOORS}
   365→
   366→# RULES
   367→- Output ONLY the slide element. No <!DOCTYPE>, no <html>, no <head>, no <body>, no <script>.
   368→- Use the template's CSS class vocabulary verbatim — do NOT invent new classes.
   369→- Use var(--accent), var(--text-1), etc. — never hardcoded colors.
   370→- Inline any slide-specific styles in a <style> block BEFORE the slide element if absolutely needed (rare).
   371→- Keep the slide visually consistent with the template's design language.
   372→- Do NOT include external <link> or <script src> tags.
   373→- Honor the content_markdown: convert it to semantic HTML using the template's classes.
   374→- Add speaker notes (${notesTag}...${notesTag.replace('<', '</').replace('>', '>')}) with the slide's speaker_notes.
   375→- No text overflow — if content doesn't fit, trim it. No animating layout properties (width/height/top/left) — transform/opacity only.
   376→
   377→# OUTPUT FORMAT
   378→Output ONLY the slide HTML element. No markdown fences. No explanation.`
   379→}

   417→export function buildScratchSystemPrompt(params: ScratchPromptParams): string {

   421→  const wordCount = vb === 'concise' ? 20 : vb === 'text-heavy' ? 60 : 40
   422→  const count = params.slideEnd - params.slideStart + 1
   423→
   424→  const engineContract = params.engine === 'reveal'
   425→    ? `# ENGINE CONTRACT (reveal.js — vendored runtime at /revealjs/)
   426→- The assembler builds the full HTML document. You output ONLY slide <section> elements (plus, on the first chunk, ONE <style> block placed before the sections).
   427→- Each slide is a plain <section>...</section> (NO class="slide", NO <div class="slides"> wrapper — the assembler adds it).
   428→- NEVER nest <section> inside <section>. One <section> per slide, flat.
   429→- Speaker notes go inside <aside class="notes">...</aside> within each section.
   430→- You MAY use reveal.js fragment syntax (<p class="fragment">) and data-background attributes.
   431→- In your <style> block define scoped helper classes used by sections: .kicker, .lede, .grid-2, .grid-3, .grid-4, .card, .big-number, .big-label, .pill, .quote, .quote-attr, .timeline, .timeline-item, .bullets — style them against reveal's CSS variables (--r-background-color, --r-main-color, --r-heading-color, --r-link-color) and your own var(--accent).

   511→/**
   512→ * Strips markdown fences / preamble and returns just the slide element
   513→ * (<section ...>, <div class="step"...> or <div class="slide"...>).
   514→ */
   515→export function extractHtml(raw: string): string {
   516→  let out = raw.trim()
   517→  const fenceMatch = out.match(/```(?:html|HTML)?\s*\n?([\s\S]*?)\n?```/)
   518→  if (fenceMatch) out = fenceMatch[1].trim()
   519→  const startIdx = out.search(/<section|<div class="step"|<div class="slide"/i)
   520→  if (startIdx >= 0) out = out.slice(startIdx)
   521→  if (out.startsWith('<section')) {
   522→    const endIdx = out.lastIndexOf('</section>')
   523→    if (endIdx >= 0) out = out.slice(0, endIdx + '</section>'.length)
   524→  } else if (out.startsWith('<div class="step')) {
   525→    const endIdx = out.lastIndexOf('</div>')
   526→    if (endIdx >= 0) out = out.slice(0, endIdx + '</div>'.length)
   527→  }
   528→  return out.trim()
   529→}

   535→/**
   536→ * Sanitizer for AI text where line breaks matter (markdown bodies):
   537→ * strips tags + control chars but preserves newlines (unlike helpers.sanitizeText).
   538→ */
   539→export function sanitizeMarkdownText(input: unknown, maxLen = 2000): string {
   540→  if (input == null) return ''
   541→  return String(input)
   542→    .replace(/<[^>]*>/g, '')
   543→    .replace(/\r\n?/g, '\n')
   544→    .replace(/[\u0000-\u0008\u000b\u000c\u000e-\u001f]/g, '')
   545→    .replace(/[ \t]+\n/g, '\n')
   546→    .replace(/\n{3,}/g, '\n\n')
   547→    .trim()
   548→    .slice(0, maxLen)
   549→}
