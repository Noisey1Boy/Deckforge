   663→export const DEFAULT_THEME_ID = 'corporate-clean';
   664→
   665→export function getTheme(id: string | undefined | null): ThemeManifest {
   666→  return THEMES.find((t) => t.id === id) ?? THEMES.find((t) => t.id === DEFAULT_THEME_ID)!;
   667→}
   668→
   669→const HEX_RE = /^#(?:[0-9a-f]{3}|[0-9a-f]{6})$/i;
   670→
   671→/** Normalize #rgb → #rrggbb so luminance math always sees 6 digits. */
   672→function expandHex(hex: string): string {
   673→  if (!/^#[0-9a-f]{6}$/i.test(hex)) {
   674→    const [r, g, b] = hex.slice(1).split('');
   675→    return `#${r}${r}${g}${g}${b}${b}`.toLowerCase();
   676→  }
   677→  return hex.toLowerCase();
   678→}
   679→
   680→/** Relative luminance (WCAG 2.1) of a #rrggbb color. */
   681→function relLuminance(hex: string): number {
   682→  const n = parseInt(expandHex(hex).slice(1), 16);
   683→  const lin = (v: number) => {
   684→    const c = v / 255;
   685→    return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4);
   686→  };
   687→  return 0.2126 * lin((n >> 16) & 255) + 0.7152 * lin((n >> 8) & 255) + 0.0722 * lin(n & 255);
   688→}
   689→
   690→/**
   691→ * Pick the most readable foreground (white vs. near-black) for text sitting
   692→ * on `hex`, per WCAG contrast ratio — keeps light user accents (butter,
   693→ * mint, sky) from pairing with white text and failing accessibility.
   694→ */
   695→function readableOn(hex: string): string {
   696→  const l = relLuminance(hex);
   697→  const contrastWhite = 1.05 / (l + 0.05);
   698→  const contrastBlack = (l + 0.05) / 0.05;
   699→  return contrastWhite >= contrastBlack ? '#ffffff' : '#101010';
   700→}
   701→
   702→/** WCAG contrast ratio between two hex colors (1–21). */
   703→export function contrastRatio(a: string, b: string): number {
   704→  const la = relLuminance(a);
   705→  const lb = relLuminance(b);
   706→  const [hi, lo] = la >= lb ? [la, lb] : [lb, la];
   707→  return (hi + 0.05) / (lo + 0.05);
   708→}
   709→
   710→/** Linear-light mix of a hex color toward black; t=0 → original, t=1 → black. */
   711→function mixTowardBlack(hex: string, t: number): string {
   712→  const n = parseInt(expandHex(hex).slice(1), 16);
   713→  const ch = (v: number) => Math.round(v * (1 - t));
   714→  const r = ch((n >> 16) & 255);
   715→  const g = ch((n >> 8) & 255);
   716→  const b = ch(n & 255);
   717→  return `#${((r << 16) | (g << 8) | b).toString(16).padStart(6, '0')}`;
   718→}
   719→
   720→/**
   721→ * Return a version of `hex` that reads as TEXT on `bg` (≥4.5:1, WCAG AA).
   722→ * Colors that already pass are returned untouched; light ones are stepped
   723→ * toward black until they pass. Used for accent-as-text (kickers, labels).
   724→ */
   725→function textSafeOn(hex: string, bg: string): string {
   726→  if (contrastRatio(hex, bg) >= 4.5) return expandHex(hex);
   727→  for (let i = 1; i <= 10; i++) {
   728→    const candidate = mixTowardBlack(hex, i / 10);
   729→    if (contrastRatio(candidate, bg) >= 4.5) return candidate;
   730→  }
   731→  return '#101010';
   732→}
   733→
   734→/**
   735→ * Layer a user-chosen accent over a theme: the primary color (titles' rule,
   736→ * bullets, stats values, chart series 1, decor) re-tints instantly while the
   737→ * rest of the theme (fonts, layout, backgrounds, motion) stays intact.
   738→ * `primaryContrast` is re-derived from the accent so on-primary text always
   739→ * passes contrast, whatever lightness the user picks, and `primaryText` is a
   740→ * darkened variant for accent-as-TEXT on light backgrounds (kickers, labels).
   741→ * Returns the manifest untouched when no valid accent is given.
   742→ */
   743→export function applyAccent(theme: ThemeManifest, accent?: string | null): ThemeManifest {
   744→  if (!accent || !HEX_RE.test(accent)) return theme;
   745→  return {
   746→    ...theme,
   747→    tokens: {
   748→      ...theme.tokens,
   749→      colors: {
   750→        ...theme.tokens.colors,
   751→        primary: accent,
   752→        accent,
   753→        primaryContrast: readableOn(accent),
   754→        primaryText: textSafeOn(accent, theme.tokens.colors.bg),
   755→      },
   756→      chart: { ...theme.tokens.chart, palette: [accent, ...theme.tokens.chart.palette.slice(1)] },
   757→    },
   758→  };
   759→}
   760→
   761→/** Resolve the effective render theme for a deck (theme + optional accent). */
   762→export function deckTheme(deck: Pick<ThemeDeck, 'themeId' | 'meta'>): ThemeManifest {
   763→  return applyAccent(getTheme(deck.themeId), deck.meta?.accent);
   764→}
   765→
   766→interface ThemeDeck {
   767→  themeId: string;
   768→  meta?: { accent?: string };
   769→}
