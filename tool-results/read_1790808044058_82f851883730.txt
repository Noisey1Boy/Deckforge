     4→import { AnimatePresence, motion, useReducedMotion } from 'framer-motion';
     5→import { useAppStore } from '@/lib/store/deck-store';
     6→import { getTheme, deckTheme } from '@/lib/design/themes';
     7→import { deckFooterText } from '@/lib/deck/types';
     8→import { SlideView } from '@/components/deck/SlideView';
     9→import { ScaledSlide } from '@/components/deck/ScaledSlide';
    10→import { Button } from '@/components/ui/button';
    11→import { DropdownMenu, DropdownMenuContent, DropdownMenuItem, DropdownMenuLabel, DropdownMenuSeparator, DropdownMenuTrigger } from '@/components/ui/dropdown-menu';
    12→import { X, ChevronLeft, ChevronRight, StickyNote, Maximize2, Minimize2, LayoutGrid, Timer, Play, Pause, ChevronsUpDown, Repeat, Projector, MonitorOff } from 'lucide-react';

    15→
    16→const AUTOPLAY_SPEEDS = [5, 10, 15, 20] as const;
    17→const LS_SPEED = 'deckforge.autoplay.speed';
    18→const LS_LOOP = 'deckforge.autoplay.loop';
    19→const LS_PRESENTER_W = 'deckforge.presenter.width';
    20→const PRESENTER_W_MIN = 280;
    21→const PRESENTER_W_MAX = 620;
    22→const PRESENTER_W_DEFAULT = 416;
    23→
    24→function loadSpeed(): number {
    25→  try {
    26→    const v = Number(window.localStorage.getItem(LS_SPEED));
    27→    return (AUTOPLAY_SPEEDS as readonly number[]).includes(v) ? v : 10;
    28→  } catch {
    29→    return 10;
    30→  }
    31→}
    32→function loadLoop(): boolean {
    33→  try {
    34→    return window.localStorage.getItem(LS_LOOP) === '1';
    35→  } catch {
    36→    return false;
    37→  }
    38→}
    39→function loadPresenterWidth(): number {
    40→  try {
    41→    const raw = window.localStorage.getItem(LS_PRESENTER_W);
    42→    if (!raw) return PRESENTER_W_DEFAULT;
    43→    const v = Number(raw);
    44→    return Number.isFinite(v) ? Math.min(PRESENTER_W_MAX, Math.max(PRESENTER_W_MIN, v)) : PRESENTER_W_DEFAULT;
    45→  } catch {
    46→    return PRESENTER_W_DEFAULT;
    47→  }
    48→}
    49→
    50→interface MotionPreset {
    51→  initial: Record<string, unknown>;
    52→  animate: Record<string, unknown>;
    53→  exit: Record<string, unknown>;
    54→}
    55→
    56→/** Framer Motion preset for a transition kind ("inherit" follows the theme). */
    57→function motionFor(kind: TransitionKind | undefined, themeStyle: string, reduceMotion: boolean): MotionPreset | null {
    58→  if (reduceMotion) {
    59→    return { initial: { opacity: 0 }, animate: { opacity: 1 }, exit: { opacity: 0 } };
    60→  }
    61→  const style = !kind || kind === 'inherit' ? themeStyle : kind;
    62→  switch (style) {
    63→    case 'none':
    64→      return null; // hard cut — handled by the caller
    65→    case 'rise':
    66→      return { initial: { opacity: 0, y: 42 }, animate: { opacity: 1, y: 0 }, exit: { opacity: 0, y: -28 } };
    67→    case 'zoom':
    68→      return { initial: { opacity: 0, scale: 0.92 }, animate: { opacity: 1, scale: 1 }, exit: { opacity: 0, scale: 1.05 } };
    69→    case 'slide':
    70→      return { initial: { opacity: 0, x: 120 }, animate: { opacity: 1, x: 0 }, exit: { opacity: 0, x: -90 } };
    71→    default:
    72→      return { initial: { opacity: 0 }, animate: { opacity: 1 }, exit: { opacity: 0 } };
    73→  }
    74→}

    76→/**
    77→ * Fullscreen present mode: keyboard navigation, click-to-seek progress bar,
    78→ * speaker notes, slide overview, auto-play with loop, a dedicated presenter
    79→ * view (notes + next-slide preview) and per-slide transition overrides —
    80→ * all transitions disabled under prefers-reduced-motion.
    81→ */
    82→export function PresentOverlay() {
    83→  const presenting = useAppStore((s) => s.presenting);
    84→  const setPresenting = useAppStore((s) => s.setPresenting);
    85→  const deck = useAppStore((s) => s.deck);
    86→  const index = useAppStore((s) => s.currentSlideIndex);
    87→  const setCurrent = useAppStore((s) => s.setCurrentSlide);
    88→  const [showNotes, setShowNotes] = useState(false);
    89→  const [showOverview, setShowOverview] = useState(false);
    90→  const [presenter, setPresenter] = useState(false);
    91→  const [blanked, setBlanked] = useState(false);
    92→  const [presenterW, setPresenterWState] = useState<number>(() => (typeof window === 'undefined' ? PRESENTER_W_DEFAULT : loadPresenterWidth()));
    93→  const [isFull, setIsFull] = useState(false);
    94→  const [scale, setScale] = useState(1);
    95→  const [elapsed, setElapsed] = useState(0);
    96→  const [autoplay, setAutoplay] = useState(false);
    97→  const [speed, setSpeedState] = useState<number>(() => (typeof window === 'undefined' ? 10 : loadSpeed()));
    98→  const [loop, setLoopState] = useState<boolean>(() => (typeof window === 'undefined' ? false : loadLoop()));
    99→  const startedAt = useRef<number | null>(null);
   100→  const presenterDrag = useRef<{ startX: number; startW: number } | null>(null);
   101→  const reduceMotion = useReducedMotion();
   102→  const setSpeed = useCallback((v: number) => {
   103→    setSpeedState(v);
   104→    try { window.localStorage.setItem(LS_SPEED, String(v)); } catch { /* private mode */ }
   105→  }, []);
   106→  const setLoop = useCallback((v: boolean) => {
   107→    setLoopState(v);
   108→    try { window.localStorage.setItem(LS_LOOP, v ? '1' : '0'); } catch { /* private mode */ }
   109→  }, []);
   110→  const setPresenterW = useCallback((v: number) => {
   111→    const clamped = Math.min(PRESENTER_W_MAX, Math.max(PRESENTER_W_MIN, Math.round(v)));
   112→    setPresenterWState(clamped);
   113→    try { window.localStorage.setItem(LS_PRESENTER_W, String(clamped)); } catch { /* private mode */ }
   114→  }, []);
   115→
   116→  const slides = deck?.slides ?? [];
   117→  const theme = useMemo(() => (deck ? deckTheme(deck) : getTheme(deck?.themeId)), [deck]);
   118→  const footerText = deck ? deckFooterText(deck) : '';
   119→  const showPageNumbers = deck?.meta?.showPageNumbers !== false;
   120→  const slide = slides[index];
   121→  const visibleCount = useMemo(() => slides.filter((s) => !s.hidden).length, [slides]);
   122→  const visiblePos = useMemo(() => slides.slice(0, index).filter((s) => !s.hidden).length, [slides, index]);
   123→  const nextSlide = useMemo(() => {
   124→    let j = index + 1;
   125→    while (j < slides.length && slides[j]?.hidden) j++;
   126→    return j < slides.length ? slides[j] : null;
   127→  }, [slides, index]);
   128→  const nextIndex = useMemo(() => {
   129→    let j = index + 1;
   130→    while (j < slides.length && slides[j]?.hidden) j++;
   131→    return j < slides.length ? j : null;
   132→  }, [slides, index]);
   133→
   134→  /** Move one slide in a direction, skipping hidden slides. */
   135→  const stepTo = useCallback(
   136→    (dir: 1 | -1) => {
   137→      let j = index + dir;
   138→      while (j >= 0 && j < slides.length && slides[j]?.hidden) j += dir;
   139→      if (j >= 0 && j < slides.length) setCurrent(j);
   140→    },
   141→    [index, slides, setCurrent]
   142→  );
   143→
   144→  const go = useCallback((delta: number) => stepTo(delta as 1 | -1), [stepTo]);
   145→
   146→  /** Exit presentation and reset transient state (autoplay, timer, blank). */
   147→  const exitPresent = useCallback(() => {
   148→    setAutoplay(false);
   149→    setElapsed(0);
   150→    setPresenter(false);
   151→    setBlanked(false);
   152→    setPresenting(false);
   153→  }, [setPresenting]);
   154→
   155→  // viewport scaling
   156→  useEffect(() => {
   157→    if (!presenting) return;
   158→    const update = () => {
   159→      const s = Math.min(window.innerWidth / 1280, window.innerHeight / 720);
   160→      setScale(s);
   161→    };
   162→    update();
   163→    window.addEventListener('resize', update);
   164→    return () => window.removeEventListener('resize', update);
   165→  }, [presenting]);
   166→
   167→  // presenter timer (starts on entry; reset happens in exitPresent)
   168→  useEffect(() => {
   169→    if (!presenting) {
   170→      startedAt.current = null;
   171→      return;
   172→    }
   173→    startedAt.current = Date.now();
   174→    const t = setInterval(() => {
   175→      if (startedAt.current) setElapsed(Math.floor((Date.now() - startedAt.current) / 1000));
   176→    }, 500);
   177→    return () => clearInterval(t);
   178→  }, [presenting]);
   179→
   180→  // never land on a hidden slide when entering presentation mode
   181→  useEffect(() => {
   182→    if (!presenting) return;
   183→    const st = useAppStore.getState();
   184→    const all = st.deck?.slides ?? [];
   185→    if (all[st.currentSlideIndex]?.hidden) {
   186→      const first = all.findIndex((s) => !s.hidden);
   187→      if (first >= 0) st.setCurrentSlide(first);
   188→    }
   189→  }, [presenting]);
   190→
   191→  // auto-play: advance every `speed` seconds, skipping hidden slides.
   192→  // At the end it loops back to the first visible slide when loop is on.
   193→  useEffect(() => {
   194→    if (!presenting || !autoplay || showOverview) return;
   195→    const t = setInterval(() => {
   196→      const st = useAppStore.getState();
   197→      const all = st.deck?.slides ?? [];
   198→      let j = st.currentSlideIndex + 1;
   199→      while (j < all.length && all[j]?.hidden) j++;
   200→      if (j < all.length) {
   201→        st.setCurrentSlide(j);
   202→      } else if (loop) {
   203→        const first = all.findIndex((s) => !s.hidden);
   204→        if (first >= 0) st.setCurrentSlide(first);
   205→      } else {
   206→        setAutoplay(false);
   207→      }
   208→    }, speed * 1000);
   209→    return () => clearInterval(t);
   210→  }, [presenting, autoplay, showOverview, speed, loop, index]);
   211→
   212→  /** Jump to the visible slide nearest a 0..1 fraction of the deck. */
   213→  const seekToFraction = useCallback(
   214→    (f: number) => {
   215→      const vis = slides.map((s, i) => ({ s, i })).filter(({ s }) => !s.hidden);
   216→      if (!vis.length) return;
   217→      const target = Math.min(vis.length - 1, Math.max(0, Math.round(f * (vis.length - 1))));
   218→      setCurrent(vis[target].i);
   219→    },
   220→    [slides, setCurrent]
   221→  );
   222→
   223→  // keyboard nav
   224→  useEffect(() => {
   225→    if (!presenting) return;
   226→    const onKey = (e: KeyboardEvent) => {
   227→      switch (e.key) {
   228→        case 'ArrowRight':
   229→        case ' ':
   230→        case 'PageDown':
   231→          e.preventDefault();
   232→          go(1);
   233→          break;
   234→        case 'ArrowLeft':
   235→        case 'PageUp':
   236→          e.preventDefault();
   237→          go(-1);
   238→          break;
   239→        case 'Home': {
   240→          const first = slides.findIndex((s) => !s.hidden);
   241→          if (first >= 0) setCurrent(first);
   242→          break;
   243→        }
   244→        case 'End': {
   245→          const last = slides.length - 1 - [...slides].reverse().findIndex((s) => !s.hidden);
   246→          if (slides[last] && !slides[last].hidden) setCurrent(last);
   247→          break;
   248→        }
   249→        case 'p':
   250→        case 'P':
   251→          setAutoplay((v) => !v);
   252→          break;
   253→        case 'v':
   254→        case 'V':
   255→          setPresenter((v) => !v);
   256→          break;
   257→        case 'b':
   258→        case 'B':
   259→          e.preventDefault();
   260→          setBlanked((v) => !v);
   261→          break;
   262→        case 'Escape': {
   263→          // let Radix menus / popovers consume Escape first (don't exit the deck)
   264→          if (document.querySelector('[role="menu"], [data-radix-popper-content-wrapper]')) break;
   265→          if (showOverview) setShowOverview(false);
   266→          else exitPresent();
   267→          break;
   268→        }
   269→        case 'n':
   270→        case 'N':
   271→          setShowNotes((v) => !v);
   272→          break;
   273→        case 'g':
   274→        case 'G':
   275→          setShowOverview((v) => !v);
   276→          break;
   277→      }
   278→    };
   279→    window.addEventListener('keydown', onKey);
   280→    return () => window.removeEventListener('keydown', onKey);
   281→  }, [presenting, go, setCurrent, exitPresent, slides.length, showOverview]);
   282→
   283→  const enterFull = useCallback(async () => {
   284→    try {
   285→      if (!document.fullscreenElement) {
   286→        await document.documentElement.requestFullscreen();
   287→        setIsFull(true);
   288→      } else {
   289→        await document.exitFullscreen();
   290→        setIsFull(false);
   291→      }
   292→    } catch {
   293→      /* fullscreen may be blocked — presenting still works */
   294→    }
   295→  }, []);
   296→
   297→  useEffect(() => {
   298→    const onFs = () => setIsFull(!!document.fullscreenElement);
   299→    document.addEventListener('fullscreenchange', onFs);
   300→    return () => document.removeEventListener('fullscreenchange', onFs);
   301→  }, []);
   302→
   303→  if (!presenting || !deck) return null;
   304→
   305→  const mm = String(Math.floor(elapsed / 60)).padStart(2, '0');
   306→  const ss = String(elapsed % 60).padStart(2, '0');
   307→
   308→  // per-slide transition override: slide.transition → theme motion style
   309→  const preset = motionFor(slide?.transition, theme.tokens.motion.style, !!reduceMotion);
   310→  const cut = !preset; // 'none' — hard cut, no enter animation
   311→
   312→  // stage (shared between standard and presenter layout)
   313→  const stage = (
   314→    <div className="flex-1 min-h-0 grid place-items-center relative overflow-hidden">
   315→      {/* subtle spotlight vignette behind the slide */}
   316→      <div
   317→        className="absolute inset-0 pointer-events-none"
   318→        style={{ background: 'radial-gradient(ellipse 62% 58% at 50% 46%, rgba(255,255,255,0.055), transparent 68%)' }}
   319→        aria-hidden
   320→      />
   321→      {/* split click zones: left 28% back, rest forward (keyboard/arrows also work) */}
   322→      <button
   323→        className="absolute inset-y-0 left-0 w-[28%] z-[1] cursor-w-resize"
   324→        onClick={() => go(-1)}
   325→        disabled={index === 0}
   326→        tabIndex={-1}
   327→        aria-hidden
   328→      />
   329→      <button
   330→        className="absolute inset-y-0 right-0 w-[72%] z-[1] cursor-e-resize"
   331→        onClick={() => go(1)}
   332→        disabled={index >= slides.length - 1}
   333→        tabIndex={-1}
   334→        aria-hidden
   335→      />
   336→      <AnimatePresence mode="wait" initial={false}>
   337→        <motion.div
   338→          key={index}
   339→          initial={cut ? false : preset!.initial}
   340→          animate={cut ? { opacity: 1 } : { ...preset!.animate, transition: { duration: theme.tokens.motion.duration, ease: [0.22, 0.61, 0.36, 1] } }}
   341→          exit={cut ? { opacity: 0, transition: { duration: 0.08 } } : { ...preset!.exit, transition: { duration: 0.18 } }}
   342→          style={{
   343→            width: 1280 * scale,
   344→            height: 720 * scale,
   345→            boxShadow: '0 24px 96px rgba(0,0,0,.5)',
   346→          }}
   347→          className="overflow-hidden rounded-sm"
   348→        >
   349→          <div style={{ width: 1280, height: 720, transform: `scale(${scale})`, transformOrigin: 'top left' }}>
   350→            {slide ? (
   351→              <SlideView slide={slide} theme={theme} deckTitle={footerText} index={index} pageCount={slides.length} showPageNumbers={showPageNumbers} />
   352→            ) : null}
   353→          </div>
   354→        </motion.div>
   355→      </AnimatePresence>
   356→    </div>
   357→  );
   358→
   359→  return (
   360→    <div className="fixed inset-0 z-[100] bg-black flex flex-col" role="dialog" aria-modal="true" aria-label="Presentation mode">
   361→      {presenter ? (
   362→        /* ------------------------------------------------ presenter view */
   363→        <div className="flex-1 min-h-0 flex flex-col lg:flex-row" style={{ '--pw': `${presenterW}px` } as React.CSSProperties}>
   364→          {stage}
   365→          {/* drag handle — resize the presenter panel (desktop) */}
   366→          <div
   367→            role="separator"
   368→            aria-orientation="vertical"
   369→            aria-label="Resize presenter panel (arrow keys adjust)"
   370→            tabIndex={0}
   371→            onPointerDown={(e) => {
   372→              e.currentTarget.setPointerCapture(e.pointerId);
   373→              presenterDrag.current = { startX: e.clientX, startW: presenterW };
   374→            }}
   375→            onPointerMove={(e) => {
   376→              const d = presenterDrag.current;
   377→              if (!d) return;
   378→              setPresenterW(d.startW - (e.clientX - d.startX));
   379→            }}
   380→            onPointerUp={() => { presenterDrag.current = null; }}
   381→            onKeyDown={(e) => {
   382→              if (e.key === 'ArrowLeft') { e.preventDefault(); setPresenterW(presenterW + 32); }
   383→              if (e.key === 'ArrowRight') { e.preventDefault(); setPresenterW(presenterW - 32); }
   384→            }}
   385→            className="hidden lg:block relative w-1.5 -mx-0.5 z-10 cursor-col-resize shrink-0 group/sep"
   386→            title="Drag to resize · arrow keys"
   387→          >
   388→            <span className="absolute inset-y-0 -left-1 -right-1" aria-hidden />
   389→            <span className="absolute inset-y-0 left-0 w-px bg-white/10 group-hover/sep:bg-white/40 group-focus-visible/sep:bg-white/60 transition-colors" aria-hidden />
   390→          </div>
   391→          <aside
   392→            className="w-full lg:w-[var(--pw,26rem)] shrink-0 border-t lg:border-t-0 lg:border-l border-white/10 bg-neutral-950/95 backdrop-blur p-4 lg:p-5 flex lg:flex-col gap-4 overflow-y-auto max-h-[46%] lg:max-h-none"
   393→            aria-label="Presenter view"
   394→          >
   395→            <div className="flex items-center justify-between lg:justify-start gap-4 shrink-0">
   396→              <div>
   397→                <div className="text-[10px] uppercase tracking-widest text-white/40">Now showing</div>
   398→                <div className="text-white/90 text-sm font-mono tabular-nums mt-0.5">
   399→                  {Math.min(visiblePos + 1, visibleCount)} <span className="text-white/40">/ {visibleCount}</span>
   400→                </div>
   401→              </div>
   402→              <div className="lg:mt-2">
   403→                <div className="text-[10px] uppercase tracking-widest text-white/40 flex items-center gap-1"><Timer className="h-3 w-3" /> Elapsed</div>
   404→                <div className="text-white/90 text-sm font-mono tabular-nums mt-0.5">{mm}:{ss}</div>
   405→              </div>
   406→              {autoplay && (
   407→                <span className="inline-flex items-center gap-1 text-[10px] uppercase tracking-wider rounded-full bg-emerald-500/15 text-emerald-300 border border-emerald-400/30 px-2 py-0.5 self-start lg:mt-2">
   408→                  <Play className="h-2.5 w-2.5" /> Auto {speed}s
   409→                </span>
   410→              )}
   411→            </div>
   412→
   413→            <div className="flex-1 min-h-20 lg:min-h-0 lg:overflow-y-auto">
   414→              <div className="text-[10px] uppercase tracking-widest text-white/40 mb-1.5">Speaker notes</div>
   415→              {slide?.notes ? (
   416→                <p className="text-white/90 text-sm lg:text-base leading-relaxed whitespace-pre-wrap">{slide.notes}</p>
   417→              ) : (
   418→                <p className="text-white/35 text-sm italic">No notes for this slide — add them in the editor inspector.</p>
   419→              )}
   420→            </div>
   421→
   422→            <div className="shrink-0 lg:mt-auto">
   423→              <div className="text-[10px] uppercase tracking-widest text-white/40 mb-1.5">Up next</div>
   424→              {nextSlide ? (
   425→                <button
   426→                  onClick={() => nextIndex != null && setCurrent(nextIndex)}
   427→                  className="block w-full rounded-md overflow-hidden border border-white/15 hover:border-white/50 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-white text-left"
   428→                  aria-label={`Preview next slide: ${nextSlide.title || nextSlide.layout}`}
   429→                >
   430→                  <ScaledSlide slide={nextSlide} theme={theme} index={nextIndex ?? undefined} pageCount={slides.length} showPageNumbers={false} />
   431→                </button>
   432→              ) : (
   433→                <div className="h-24 grid place-items-center rounded-md border border-dashed border-white/15 text-white/35 text-xs">
   434→                  End of deck
   435→                </div>
   436→              )}
   437→            </div>
   438→          </aside>
   439→        </div>
   440→      ) : (
   441→        /* ---------------------------------------------- standard layout */
   442→        stage
   443→      )}
   444→
   445→      {/* progress — paced over visible slides; click / drag to seek */}
   446→      <div
   447→        className="group h-1.5 hover:h-3 transition-[height] duration-200 bg-neutral-800 shrink-0 cursor-pointer relative"
   448→        role="slider"
   449→        aria-label="Seek to slide"
   450→        aria-valuemin={1}
   451→        aria-valuemax={Math.max(visibleCount, 1)}
   452→        aria-valuenow={Math.min(visiblePos + 1, visibleCount)}
   453→        tabIndex={0}
   454→        onKeyDown={(e) => {
   455→          if (e.key === 'ArrowRight' || e.key === 'ArrowUp') { e.preventDefault(); go(1); }
   456→          if (e.key === 'ArrowLeft' || e.key === 'ArrowDown') { e.preventDefault(); go(-1); }
   457→        }}
   458→        onPointerDown={(e) => {
   459→          const rect = e.currentTarget.getBoundingClientRect();
   460→          seekToFraction((e.clientX - rect.left) / rect.width);
   461→        }}
   462→      >
   463→        <div
   464→          className="h-full transition-all duration-300 pointer-events-none"
   465→          style={{ width: `${((visiblePos + 1) / Math.max(visibleCount, 1)) * 100}%`, background: `linear-gradient(90deg, ${theme.tokens.colors.primary}, ${theme.tokens.colors.accent})` }}
   466→        />
   467→      </div>
   468→
   469→      {/* blank screen (projector shutter) — B key or click to restore */}
   470→      {blanked && (
   471→        <div
   472→          className="absolute inset-0 z-[2] bg-black grid place-items-center cursor-pointer"
   473→          onClick={() => setBlanked(false)}
   474→          role="button"
   475→          aria-label="Screen hidden. Click or press B to show the slide"
   476→        >
   477→          <p className="text-white/30 text-sm">
   478→            Screen hidden — press <span className="font-mono text-white/50">B</span> or click to show
   479→          </p>
   480→        </div>
   481→      )}
   482→
   483→      {/* transient keyboard hint (CSS-animated: fades out once per entry) */}
   484→      {presenting && !showOverview && (
   485→        <div

   491→        </div>

   493→
   494→      {/* controls */}

   540→              </DropdownMenuItem>
   541→            ))}
   542→            <DropdownMenuSeparator />
   543→            <DropdownMenuItem
   544→              onClick={() => setLoop((v) => !v)}
   545→              className="justify-between text-sm"
   546→              role="menuitemcheckbox"
   547→              aria-checked={loop}
   548→            >

   559→        <Button variant="ghost" size="icon" className={cn('h-8 w-8 text-white hover:bg-white/15', showOverview && 'bg-white/15')} onClick={() => setShowOverview((v) => !v)} aria-label="Slide overview (G)">

   596→      <AnimatePresence>
   597→        {showNotes && !presenter && slide?.notes && (

   601→            exit={{ opacity: 0, y: 24 }}

   605→            <div className="text-[10px] uppercase tracking-widest text-white/50 mb-1.5">Speaker notes · N to toggle</div>

   610→

   632→                {slides.map((s, i) => ({ s, i })).filter(({ s }) => !s.hidden).map(({ s, i }, vi) => (

   648→            </div>
