   530→export function PitchDeck() {
   531→  const [index, setIndex] = useState(0);
   532→  const [dir, setDir] = useState(1);
   533→  const [elapsed, setElapsed] = useState(0);
   534→  const [started, setStarted] = useState(false);
   535→  const stageRef = useRef<HTMLDivElement | null>(null);
   536→  const reduceMotion = useReducedMotion();
   537→
   538→  const go = useCallback((next: number) => {
   539→    setIndex((cur) => {
   540→      const clamped = Math.max(0, Math.min(SLIDE_COUNT - 1, next));
   541→      if (clamped !== cur) setDir(clamped > cur ? 1 : -1);
   542→      setStarted(true);
   543→      return clamped;
   544→    });
   545→  }, []);
   546→
   547→  const slides: Slide[] = useMemo(() => buildSlides(go), [go]);
   548→
   549→  // keyboard
   550→  useEffect(() => {
   551→    const onKey = (e: KeyboardEvent) => {
   552→      if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'PageDown') { e.preventDefault(); go(index + 1); }
   553→      else if (e.key === 'ArrowLeft' || e.key === 'PageUp') { e.preventDefault(); go(index - 1); }
   554→      else if (e.key === 'Home') go(0);
   555→      else if (e.key === 'End') go(slides.length - 1);
   556→      else if (e.key.toLowerCase() === 'f') {
   557→        try {
   558→          if (document.fullscreenElement) void document.exitFullscreen();
   559→          else void stageRef.current?.requestFullscreen();
   560→        } catch { /* denied */ }
   561→      }
   562→    };
   563→    window.addEventListener('keydown', onKey);
   564→    return () => window.removeEventListener('keydown', onKey);
   565→  }, [go, index, slides.length]);
   566→
   567→  // pacing timer
   568→  useEffect(() => {
   569→    if (!started) return;
   570→    const id = setInterval(() => setElapsed((s) => s + 1), 1000);
   571→    return () => clearInterval(id);
   572→  }, [started]);
   573→
   574→  const over = elapsed > TARGET_SECONDS;
   575→  const mm = Math.floor(elapsed / 60);
   576→  const ss = String(elapsed % 60).padStart(2, '0');
   577→
   578→  return (
   579→    <div ref={stageRef} className="ps-scope fixed inset-0 flex flex-col bg-[color:var(--stg-cream)] text-[color:var(--stg-ink)]" data-testid="pitch-deck">
   580→      {/* top chrome */}
   581→      <header className="flex h-12 shrink-0 items-center gap-3 border-b-2 border-[color:var(--stg-ink)] bg-[color:var(--stg-ink)] px-4 text-[color:var(--stg-cream)] sm:px-6">
   582→        <Link href="/" className="stg-label text-[9px] text-white/60 transition-colors hover:text-white" aria-label="Back to DeckForge home">← DECKFORGE</Link>
   583→        <span className="stg-label hidden text-[9px] text-[color:var(--ps-link-dark)] sm:block">IDEA PITCH · THE VOICE AGENT PLATFORM</span>
   584→        <span className={cn('stg-label ml-auto flex items-center gap-1.5 rounded-full border px-3 py-1 text-[9px] tabular-nums', over ? 'border-[color:var(--ps-warn)] text-[color:var(--ps-warn)]' : 'border-white/25 text-white/70')} data-testid="pitch-timer">
   585→          <Clock className="h-3 w-3" aria-hidden /> {mm}:{ss} / 4:00 {over ? '· OVER' : ''}
   586→        </span>
   587→        <button onClick={() => { try { if (document.fullscreenElement) void document.exitFullscreen(); else void stageRef.current?.requestFullscreen(); } catch { /* denied */ } }}
   588→          aria-label="Toggle fullscreen" className="grid h-8 w-8 place-items-center text-white/60 transition-colors hover:text-white">
   589→          <Maximize className="h-4 w-4" aria-hidden />
   590→        </button>
   591→      </header>
   592→
   593→      {/* stage */}
   594→      <main className="relative min-h-0 flex-1 overflow-hidden">
   595→        <AnimatePresence mode="wait" custom={dir} initial={false}>
   596→          <motion.section
   597→            key={slides[index].id}
   598→            custom={dir}
   599→            variants={reduceMotion ? { enter: { opacity: 0 }, center: { opacity: 1 }, exit: { opacity: 0 } } : variants}
   600→            initial="enter" animate="center" exit="exit"
   601→            transition={{ duration: reduceMotion ? 0.12 : 0.34, ease: [0.16, 1, 0.3, 1] }}
   602→            className="absolute inset-0 flex flex-col"
   603→            data-testid={`pitch-slide-${slides[index].id}`}
   604→          >
   605→            {slides[index].node}
   606→          </motion.section>
   607→        </AnimatePresence>
   608→      </main>
   609→
   610→      {/* bottom chrome */}
