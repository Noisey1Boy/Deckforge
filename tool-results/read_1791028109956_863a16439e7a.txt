   184→   184→        r.onerror = reject;
   185→   185→        r.readAsDataURL(file);

   195→   195→      toast.success('Chart extracted from image.');

   392→   392→      const dt = new DataTransfer();
   393→   393→      images.forEach((f) => dt.items.add(f));

   397→   397→
