    41→/** Pick `count` example prompts starting from a rotating seed. */
    42→function dealExamples(seed: number, count = 4): string[] {
    43→  const start = (seed * count) % EXAMPLES.length;
    44→  return Array.from({ length: Math.min(count, EXAMPLES.length) }, (_, i) => EXAMPLES[(start + i) % EXAMPLES.length]);
    45→}

    52→export function CreateView() {

    54→  const setView = useAppStore((s) => s.setView);

    78→  // (avoids a hydration mismatch); deal a random set right after mount.

    81→    const t = window.setTimeout(() => setIdeaSeed(Math.floor(Math.random() * 3)), 0);

    88→  const [diagram, setDiagram] = useState<MermaidPlan | null>(null);
    89→  const [mermaid, setMermaid] = useState('');
    90→  const [markdown, setMarkdown] = useState('');
    91→  const [dataError, setDataError] = useState<string | null>(null);
    92→  const fileRef = useRef<HTMLInputElement>(null);
    93→  const imgRef = useRef<HTMLInputElement>(null);
    94→
    95→  // ---------------------------------------------------------------- data
    96→  const handleFiles = async (files: FileList | null) => {
    97→    if (!files?.length) return;
    98→    setDataError(null);
    99→    setBusy('Reading data…');
   100→    const newTables: ParsedTable[] = [...tables];
   101→    for (const file of Array.from(files).slice(0, 3)) {
   102→      const { tables: ts, error } = await parseFileToTables(file);
   103→      if (error) {
   104→        setDataError(error);
   105→        toast.error(error);
   106→      }
   107→      newTables.push(...ts);
   108→    }
   109→    setTables(newTables.slice(0, 3));
   110→    setChartSpecs([]); // re-plan after new data
   111→    setBusy(null);
   112→  };
   113→
   114→  const handleChartImage = async (files: FileList | null) => {
   115→    const file = files?.[0];
   116→    if (!file) return;
   117→    if (!/^image\/(png|jpe?g|webp)$/.test(file.type)) {
   118→      toast.error('Use a PNG, JPG or WebP screenshot of your chart.');
   119→      return;
   120→    }
   121→    if (file.size > 6_000_000) {
   122→      toast.error('Image must be under 6 MB.');
   123→      return;
   124→    }
   125→    setBusy('Reading chart from image…');
   126→    try {
   127→      const dataUrl = await new Promise<string>((resolve, reject) => {
   128→        const r = new FileReader();
   129→        r.onload = () => resolve(String(r.result));
   130→        r.onerror = reject;
   131→        r.readAsDataURL(file);
   132→      });
   133→      const res = await fetch('/api/data/chart-image', {
   134→        method: 'POST',
   135→        headers: { 'Content-Type': 'application/json' },
   136→        body: JSON.stringify({ image: dataUrl, language }),
   137→      });
   138→      const json = await res.json();
   139→      if (!res.ok) throw new Error(json.error || 'Extraction failed');
   140→      setChartSpecs((prev) => [...prev, json.chart as ChartSpec]);
   141→      toast.success('Chart extracted from image.');
   142→    } catch (e) {
   143→      toast.error(e instanceof Error ? e.message : 'Image analysis failed');
   144→    } finally {
   145→      setBusy(null);
   146→      if (imgRef.current) imgRef.current.value = '';
   147→    }
   148→  };
   149→
   150→  const analyzeMermaid = async () => {
   151→    if (mermaid.trim().length < 10) {
   152→      toast.error('Paste a Mermaid diagram first (e.g. "graph LR; A-->B;").');
   153→      return;
   154→    }
   155→    setBusy('Converting diagram…');
   156→    try {
   157→      const res = await fetch('/api/data/analyze', {
   158→        method: 'POST',
   159→        headers: { 'Content-Type': 'application/json' },
   160→        body: JSON.stringify({ mermaid, language }),
   161→      });
   162→      const json = await res.json();
   163→      if (!res.ok) throw new Error(json.error || 'Conversion failed');
   164→      if (json.diagram) {
   165→        setDiagram(json.diagram as MermaidPlan);
   166→        toast.success('Diagram ready to include.');
   167→      } else throw new Error('Could not parse the diagram');
   168→    } catch (e) {
   169→      toast.error(e instanceof Error ? e.message : 'Conversion failed');
   170→    } finally {
   171→      setBusy(null);
   172→    }
   173→  };
   174→
   175→  // ------------------------------------------------------------- generate
   176→  const generateOutline = async () => {
   177→    if (prompt.trim().length < 3) {
   178→      toast.error('Describe what the presentation should be about.');
   179→      return;
   180→    }
   181→    setBusy('Designing the outline…');
   182→    try {
   183→      // Persist generation controls so the slide pipeline uses the same settings.
   184→      setGenSettings({ audience, tone, language, slideCount });
   185→      let charts: ChartSpec[] = [...chartSpecs];
   186→      let diagramPlan: MermaidPlan | undefined = diagram ?? undefined;
   187→      let dataContext: string | undefined;
   188→
   189→      const wantsAnalysis = tables.length > 0;
   190→      if (wantsAnalysis) {
   191→        setBusy('Analyzing your data…');
   192→        const res = await fetch('/api/data/analyze', {
   193→          method: 'POST',
   194→          headers: { 'Content-Type': 'application/json' },
   195→          body: JSON.stringify({ tables: tables.map(tableToMeta), language }),
   196→        });
   197→        const json = await res.json();
   198→        if (!res.ok) throw new Error(json.error || 'Analysis failed');
   199→        const plans = (json.charts ?? []) as ChartPlan[];
   200→        const specs = plans
   201→          .map((p) => {
   202→            const t = tables[p.tableIndex];
   203→            return t ? buildChartSpec(t, p) : null;
   204→          })
   205→          .filter((c): c is ChartSpec => !!c);
   206→        charts = [...charts, ...specs];
   207→        dataContext = json.dataContext;
   208→      }
   209→      if (diagramPlan) {
   210→        dataContext = `${dataContext ?? ''}\nA Mermaid flow diagram "${diagramPlan.title}" with ${diagramPlan.nodes.length} steps is available for a "diagram" slide.`;
   211→      }
   212→
   213→      setBusy('Writing the outline…');
   214→      const res = await fetch('/api/generate/outline', {
   215→        method: 'POST',
   216→        headers: { 'Content-Type': 'application/json' },
   217→        body: JSON.stringify({
   218→          prompt, audience, tone, slideCount, language, themeChoice, dataContext,
   219→        }),
   220→      });
   221→      const json = await res.json();
   222→      if (!res.ok) throw new Error(json.error || 'Outline failed');
   223→      const outline = json.outline as DeckOutline;
   224→
   225→      // attach data: first N chart slides in the outline get real chart specs
   226→      const chartAssignments: Record<string, ChartSpec> = {};
   227→      const pool = [...charts];
   228→      let remainingDiagram = diagramPlan;
   229→      outline.slides.forEach((s, i) => {
   230→        if (s.layout === 'chart' && pool.length) {
   231→          chartAssignments[String(i)] = pool.shift()!;
   232→        }
   233→      });
   234→      // if diagram exists but no diagram slide planned, convert the last chart-less content slide? Keep: AI was told about it.
   235→
   236→      setDataAssets({
   237→        tables,
   238→        charts,
   239→        diagram: remainingDiagram ? { ...remainingDiagram } : undefined,
   240→        dataContext,
   241→        chartAssignments,
   242→      });
   243→      setOutline(outline);
   244→      setView('outline');
   245→    } catch (e) {
   246→      toast.error(e instanceof Error ? e.message : 'Generation failed');
   247→    } finally {
   248→      setBusy(null);
   249→    }
   250→  };
   251→

   253→  const buildFromMarkdown = () => {
   254→    if (markdown.trim().length < 10) {
   255→      toast.error('Paste some markdown first — headings become slides.');
   256→      return;
   257→    }
   258→    try {
   259→      const parsed = parseMarkdownOutline(markdown);
   260→      if (parsed.slides.length < 1) throw new Error('No headings found');

   262→      setGenSettings({ audience, tone, language, slideCount });

   264→        title: parsed.title,
   265→        subtitle: parsed.subtitle || prompt.trim().slice(0, 110),
   266→        slides: parsed.slides,

   270→        outline.recommendedThemeId = suggestThemeId(markdown);
   271→        outline.themeRationale = 'Matched to your markdown content.';
   272→      }
   273→      // attach user charts to slides tagged [chart], in order
   274→      const chartAssignments: Record<string, ChartSpec> = {};
   275→      const pool = [...chartSpecs];
   276→      outline.slides.forEach((s: OutlineItem, i: number) => {
   277→        if (s.layout === 'chart' && pool.length) {
   278→          chartAssignments[String(i)] = pool.shift()!;

   280→      });
   281→      setDataAssets({
   282→        tables, charts: chartSpecs,
   283→        diagram: diagram ?? undefined,
   284→        dataContext: undefined,
   285→        chartAssignments,
   286→      });
   287→      setOutline(outline);
   288→      setView('outline');
   289→      toast.success(`Outline built — ${outline.slides.length} slides from your markdown.`);

   291→      toast.error(e instanceof Error ? e.message : 'Could not parse markdown');
   292→    }

   295→  const openDeck = async (d: { id: string; title: string }) => {

   297→      setBusy('Opening deck…');
   298→      const res = await fetch(`/api/decks/${d.id}`);
   299→      const json = await res.json();
   300→      if (!res.ok) throw new Error(json.error || 'Failed to open');
   301→      setDeck(json.deck);
   302→      setView('editor');
   303→    } catch (e) {
   304→      toast.error(e instanceof Error ? e.message : 'Failed to open deck');
   305→    } finally {
   306→      setBusy(null);
   307→    }
   308→  };
   309→
   310→  const hasData = tables.length > 0 || chartSpecs.length > 0 || diagram;
   311→
   312→  // ---- drop a data file anywhere on the page to attach it ---------------
   313→  const [droppingData, setDroppingData] = useState(false);
   314→  const pageDrag = (e: React.DragEvent) => {
   315→    if (!e.dataTransfer.types.includes('Files')) return;
   316→    e.preventDefault();
   317→    setDroppingData(true);
   318→  };
   319→  const pageDragLeave = (e: React.DragEvent) => {
   320→    if (e.currentTarget.contains(e.relatedTarget as Node)) return;
   321→    setDroppingData(false);
   322→  };
   323→  const pageDrop = (e: React.DragEvent) => {
   324→    if (!e.dataTransfer.types.includes('Files')) return;
   325→    e.preventDefault();
   326→    setDroppingData(false);
   327→    const files = Array.from(e.dataTransfer.files);
   328→    const images = files.filter((f) => f.type.startsWith('image/'));
   329→    const data = files.filter((f) => !f.type.startsWith('image/'));
   330→    if (data.length > 0) {
   331→      setDataTab('files');
   332→      const dt = new DataTransfer();
   333→      data.forEach((f) => dt.items.add(f));
   334→      void handleFiles(dt.files);
   335→    }
   336→    if (images.length > 0) {
   337→      setDataTab('image');
   338→      const dt = new DataTransfer();
   339→      images.forEach((f) => dt.items.add(f));
   340→      void handleChartImage(dt.files);
   341→    }
   342→  };
   343→
   344→  return (
   345→    <div
   346→      className="w-full max-w-6xl mx-auto px-4 sm:px-6 py-8 sm:py-12 space-y-12"
   347→      onDragOver={pageDrag}
   348→      onDragLeave={pageDragLeave}
   349→      onDrop={pageDrop}
   350→    >
   351→      {droppingData && (
   352→        <div className="fixed inset-2 z-40 grid place-items-center rounded-2xl border-2 border-dashed border-neutral-900 bg-white/80 backdrop-blur-sm pointer-events-none" aria-hidden>
   353→          <div className="text-center">
   354→            <FileUp className="h-8 w-8 mx-auto text-neutral-700" />
   355→            <p className="mt-2 text-base font-semibold text-neutral-900">Drop your data</p>
   356→            <p className="text-xs text-muted-foreground mt-0.5">CSV · XLSX · JSON attach as data — images attach as chart screenshots</p>
   357→          </div>
   358→        </div>
   359→      )}
   360→      {/* Hero */}
   361→      <motion.header {...enter(0)} className="text-center space-y-5 max-w-3xl mx-auto relative">
   362→        {/* drifting gradient orbs */}
   363→        <div className="absolute -top-16 left-1/2 -translate-x-1/2 -z-10 w-[42rem] max-w-none h-72 overflow-visible" aria-hidden>
   364→          <span className="df-orb left-[6%] top-2 h-56 w-56 bg-teal-300/70" />
   365→          <span className="df-orb df-orb-2 left-[38%] -top-6 h-64 w-64 bg-amber-200/80" />
   366→          <span className="df-orb df-orb-3 right-[4%] top-6 h-52 w-52 bg-violet-300/60" />
   367→        </div>
   368→        <div
   369→          className="absolute inset-x-0 -top-6 h-64 -z-10 opacity-60"
   370→          style={{
   371→            backgroundImage: 'radial-gradient(#d4d4d4 1.2px, transparent 1.6px)',
   372→            backgroundSize: '22px 22px',
   373→            maskImage: 'radial-gradient(ellipse 60% 60% at 50% 35%, black 30%, transparent 75%)',
   374→            WebkitMaskImage: 'radial-gradient(ellipse 60% 60% at 50% 35%, black 30%, transparent 75%)',
   375→          }}
   376→          aria-hidden
   377→        />
   378→        <Badge variant="outline" className="gap-1.5 px-3 py-1 text-xs bg-white/70">
   379→          <Sparkles className="h-3.5 w-3.5 text-amber-500" /> AI presentation studio
   380→        </Badge>
   381→        <h1 className="text-4xl sm:text-5xl font-bold tracking-tight text-neutral-900">
   382→          Prompt to <span className="df-gradient-text">polished deck</span> in minutes
   383→        </h1>
   384→        <p className="text-muted-foreground text-lg">
   385→          Describe your topic or drop your data. DeckForge writes, structures and designs the
   386→          whole presentation — then lets you edit, present and export it.
   387→        </p>
   388→        <div className="flex flex-wrap items-center justify-center gap-2 sm:gap-3 pt-1" aria-label="How it works">
   389→          {[
   390→            { icon: <PenLine className="h-3.5 w-3.5" />, label: 'Prompt' },
   391→            { icon: <LayoutTemplate className="h-3.5 w-3.5" />, label: 'Design' },
   392→            { icon: <MonitorPlay className="h-3.5 w-3.5" />, label: 'Present' },
   393→            { icon: <Share2 className="h-3.5 w-3.5" />, label: 'Export' },
   394→          ].map((step, i, arr) => (
   395→            <span key={step.label} className="flex items-center gap-2 sm:gap-3">
   396→              <span className="inline-flex items-center gap-1.5 rounded-full border border-neutral-200 bg-white px-3 py-1.5 text-xs font-medium text-neutral-700 shadow-sm">
   397→                {step.icon} {step.label}
   398→              </span>
   399→              {i < arr.length - 1 && <span className="h-px w-4 sm:w-6 bg-neutral-300" aria-hidden />}
   400→            </span>
   401→          ))}
   402→        </div>
   403→      </motion.header>
   404→
   405→      {/* Prompt card */}
   406→      <motion.div {...enter(0.08)}>
   407→      <Card className="border-neutral-200 shadow-xl shadow-neutral-900/5">
   408→        <CardContent className="p-4 sm:p-6 space-y-5">
   409→          <Textarea
   410→            value={prompt}
   411→            onChange={(e) => setPrompt(e.target.value)}
   412→            placeholder='e.g. "Create me a presentation about our 2026 sustainability strategy for investors"'
   413→            className="min-h-24 text-base resize-none border-neutral-300 focus-visible:ring-neutral-900"
   414→            maxLength={1200}
   415→            aria-label="Presentation topic"
   416→          />
   417→          <div className="flex flex-wrap items-center gap-2">
   418→            {dealExamples(ideaSeed).map((ex) => (
   419→              <button
   420→                key={ex}
   421→                onClick={() => setPrompt(ex)}
   422→                className="text-xs rounded-full border border-neutral-200 px-3 py-1.5 text-muted-foreground hover:border-neutral-900 hover:text-neutral-900 transition-colors truncate max-w-full"
   423→              >
   424→                {ex.length > 52 ? ex.slice(0, 52) + '…' : ex}
   425→              </button>
   426→            ))}
   427→            <button
   428→              onClick={() => setIdeaSeed((s) => s + 1)}
   429→              className="inline-flex items-center gap-1.5 text-xs rounded-full border border-dashed border-neutral-300 px-3 py-1.5 text-neutral-500 hover:border-neutral-900 hover:text-neutral-900 transition-colors focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-neutral-900"
   430→              title="Show other starter ideas"
   431→              aria-label="Show other starter ideas"
   432→            >
   433→              <Dices className="h-3.5 w-3.5" /> More ideas
   434→            </button>
   435→          </div>
   436→
   437→          <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">
   438→            <label className="space-y-1.5">
   439→              <span className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Audience</span>
   440→              <Input value={audience} onChange={(e) => setAudience(e.target.value)} placeholder="investors, execs…" maxLength={120} />
   441→            </label>
   442→            <label className="space-y-1.5">
   443→              <span className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Tone</span>

   445→            </label>
   446→            <label className="space-y-1.5">
   447→              <span className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Language</span>
