   395→export function PitchDeck() {
   396→  const [index, setIndex] = useState(0);
   397→  const [dir, setDir] = useState(1);
   398→  const [elapsed, setElapsed] = useState(0);
   399→  const [started, setStarted] = useState(false);
   400→  const stageRef = useRef<HTMLDivElement | null>(null);
   401→  const reduceMotion = useReducedMotion();
   402→
   403→  const go = useCallback((next: number) => {
   404→    setIndex((cur) => {
   405→      const clamped = Math.max(0, Math.min(9, next)); // 10 slides, fixed
   406→      if (clamped !== cur) setDir(clamped > cur ? 1 : -1);
   407→      setStarted(true);
   408→      return clamped;
   409→    });
   410→  }, []);
   411→
   412→  const slides: Slide[] = useMemo(() => buildSlides(go), [go]);
   413→
   414→  // keyboard
   415→  useEffect(() => {
   416→    const onKey = (e: KeyboardEvent) => {
   417→      if (e.key === 'ArrowRight' || e.key === ' ' || e.key === 'PageDown') { e.preventDefault(); go(index + 1); }
   418→      else if (e.key === 'ArrowLeft' || e.key === 'PageUp') { e.preventDefault(); go(index - 1); }
   419→      else if (e.key === 'Home') go(0);
   420→      else if (e.key === 'End') go(slides.length - 1);
   421→      else if (e.key.toLowerCase() === 'f') {
   422→        try {
   423→          if (document.fullscreenElement) void document.exitFullscreen();
   424→          else void stageRef.current?.requestFullscreen();
   425→        } catch { /* denied */ }
   426→      }
   427→    };
   428→    window.addEventListener('keydown', onKey);
   429→    return () => window.removeEventListener('keydown', onKey);
   430→  }, [go, index, slides.length]);
   431→
   432→  // pacing timer
   433→  useEffect(() => {
   434→    if (!started) return;
   435→    const id = setInterval(() => setElapsed((s) => s + 1), 1000);
   436→    return () => clearInterval(id);
   437→  }, [started]);
   438→
   439→  const over = elapsed > TARGET_SECONDS;
   440→  const mm = Math.floor(elapsed / 60);
   441→  const ss = String(elapsed % 60).padStart(2, '0');
   442→
   443→  return (
   444→    <div ref={stageRef} className="fixed inset-0 flex flex-col bg-[color:var(--stg-cream)] text-[color:var(--stg-ink)]" data-testid="pitch-deck">
   445→      {/* top chrome */}
   446→      <header className="flex h-12 shrink-0 items-center gap-3 border-b-2 border-[color:var(--stg-ink)] bg-[color:var(--stg-ink)] px-4 text-[color:var(--stg-cream)] sm:px-6">
   447→        <Link href="/" className="stg-label text-[9px] text-white/60 transition-colors hover:text-white" aria-label="Back to DeckForge home">← DECKFORGE</Link>
   448→        <span className="stg-label hidden text-[9px] text-[color:var(--stg-yellow)] sm:block">IDEA PITCH · THE VOICE AGENT PLATFORM</span>
   449→        <span className={cn('stg-label ml-auto flex items-center gap-1.5 rounded-full border px-3 py-1 text-[9px] tabular-nums', over ? 'border-red-400/60 text-red-300' : 'border-white/25 text-white/70')} data-testid="pitch-timer">
   450→          <Clock className="h-3 w-3" aria-hidden /> {mm}:{ss} / 4:00 {over ? '· OVER' : ''}
   451→        </span>
   452→        <button onClick={() => { try { if (document.fullscreenElement) void document.exitFullscreen(); else void stageRef.current?.requestFullscreen(); } catch { /* denied */ } }}
   453→          aria-label="Toggle fullscreen" className="grid h-8 w-8 place-items-center text-white/60 transition-colors hover:text-white">
   454→          <Maximize className="h-4 w-4" aria-hidden />
   455→        </button>
   456→      </header>
   457→
   458→      {/* stage */}
   459→      <main className="relative min-h-0 flex-1 overflow-hidden">
   460→        <AnimatePresence mode="wait" custom={dir} initial={false}>
   461→          <motion.section
   462→            key={slides[index].id}
   463→            custom={dir}
   464→            variants={reduceMotion ? { enter: { opacity: 0 }, center: { opacity: 1 }, exit: { opacity: 0 } } : variants}
   465→            initial="enter" animate="center" exit="exit"
   466→            transition={{ duration: reduceMotion ? 0.12 : 0.34, ease: [0.16, 1, 0.3, 1] }}
   467→            className="absolute inset-0 flex flex-col"
   468→            data-testid={`pitch-slide-${slides[index].id}`}
   469→          >
   470→            {slides[index].node}
   471→          </motion.section>
   472→        </AnimatePresence>
   473→      </main>
   474→
   475→      {/* bottom chrome */}
   476→      <footer className="flex h-14 shrink-0 items-center gap-3 border-t-2 border-[color:var(--stg-ink)] bg-[color:var(--stg-cream-2)] px-4 sm:px-6">
   477→        <button onClick={() => go(index - 1)} disabled={index === 0} aria-label="Previous slide"
   478→          className="grid h-10 w-10 place-items-center border-2 border-[color:var(--stg-ink)] bg-white transition-all hover:bg-[color:var(--stg-yellow)] disabled:opacity-30">
   479→          <ArrowLeft className="h-4 w-4" aria-hidden />
   480→        </button>
   481→        <div className="flex flex-1 flex-wrap items-center gap-1.5" role="tablist" aria-label="Slides">
   482→          {slides.map((s, i) => (
   483→            <button key={s.id} role="tab" aria-selected={i === index} aria-label={`Slide ${i + 1}: ${s.label}`} onClick={() => go(i)}
   484→              className={cn('h-2.5 rounded-full transition-all', i === index ? 'w-7 bg-[color:var(--stg-ink)]' : 'w-2.5 bg-[color:var(--stg-ink)]/25 hover:bg-[color:var(--stg-ink)]/50')} />
   485→          ))}
   486→        </div>
   487→        <span className="stg-label hidden text-[9px] text-[color:var(--stg-ink)]/60 md:block">{slides[index].label} · ~{slides[index].budget >= 60 ? `${Math.floor(slides[index].budget / 60)}:${String(slides[index].budget % 60).padStart(2, '0')}` : `${slides[index].budget}s`}</span>
   488→        <span className="stg-label text-[9px] tabular-nums text-[color:var(--stg-ink)]/70">{String(index + 1).padStart(2, '0')} / {slides.length}</span>
   489→        <button onClick={() => go(index + 1)} disabled={index === slides.length - 1} aria-label="Next slide"
   490→          className="grid h-10 w-10 place-items-center border-2 border-[color:var(--stg-ink)] bg-[color:var(--stg-yellow)] transition-all hover:-translate-y-0.5 disabled:opacity-30">
   491→          <ArrowRight className="h-4 w-4" aria-hidden />
   492→        </button>
   493→      </footer>
   494→    </div>
   495→  );
   496→}
