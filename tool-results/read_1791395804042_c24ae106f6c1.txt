   221→   221→    function go(n, fromRemote){
   222→   222→      n = Math.max(0, Math.min(total-1, n));
   223→   223→      slides.forEach((s,i) => {
   224→   224→        s.classList.toggle('is-active', i===n);
   225→   225→        s.classList.toggle('is-prev', i<n);
   226→   226→      });
   227→   227→      idx = n;
   228→   228→      barFill.style.width = ((n+1)/total*100)+'%';
   229→   229→      const numEl = document.querySelector('.slide-number');
   230→   230→      if (numEl) { numEl.setAttribute('data-current', n+1); numEl.setAttribute('data-total', total); }
   231→   231→
   232→   232→      // notes (bottom overlay)
   233→   233→      const note = slides[n].querySelector('.notes, aside.notes, .speaker-notes');
   234→   234→      notes.innerHTML = note ? note.innerHTML : '';
   235→   235→
   236→   236→      // hash
   237→   237→      const hashTarget = '#/'+(n+1);
   238→   238→      if (location.hash !== hashTarget && !isPresenterWindow) {
   239→   239→        history.replaceState(null,'', hashTarget);
   240→   240→      }
   241→   241→
   242→   242→      // re-trigger entry animations
   243→   243→      slides[n].querySelectorAll('[data-anim]').forEach(el => {
   244→   244→        const a = el.getAttribute('data-anim');
   245→   245→        el.classList.remove('anim-'+a);
   246→   246→        void el.offsetWidth;
   247→   247→        el.classList.add('anim-'+a);
   248→   248→      });
   249→   249→
   250→   250→      // counter-up
   251→   251→      slides[n].querySelectorAll('.counter').forEach(el => {
   252→   252→        const target = parseFloat(el.getAttribute('data-to')||el.textContent);
   253→   253→        const dur = parseInt(el.getAttribute('data-dur')||'1200',10);
   254→   254→        const start = performance.now();
   255→   255→        const from = 0;
   256→   256→        function tick(now){
   257→   257→          const t = Math.min(1,(now-start)/dur);
   258→   258→          const v = from + (target-from)*(1-Math.pow(1-t,3));
   259→   259→          el.textContent = (target % 1 === 0) ? Math.round(v) : v.toFixed(1);
   260→   260→          if (t<1) requestAnimationFrame(tick);
   261→   261→        }
   262→   262→        requestAnimationFrame(tick);
   263→   263→      });
   264→   264→
   265→   265→      // Broadcast to other window (audience ↔ presenter)
   266→   266→      if (!fromRemote && bc) {
   267→   267→        bc.postMessage({ type: 'go', idx: n });
   268→   268→      }
   269→   269→    }
