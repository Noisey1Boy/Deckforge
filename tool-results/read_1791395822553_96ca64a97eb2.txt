   306→export function buildSystemPrompt(params: GenerateParams): string {
   307→  const audience = params.audience?.trim() || 'general business audience'
   308→  const tone = params.tone?.trim() || 'professional, confident, clear'
   309→  const focus = params.focus?.trim() || 'general purpose'
   310→
   311→  // Detect framework from template HTML
   312→  const html = params.templateHtml
   313→  let framework: 'open-design' | 'reveal.js' | 'impress.js' | 'shower' | 'dzslides'
   314→  if (/class="reveal"|Reveal\.initialize|\/revealjs\//i.test(html)) framework = 'reveal.js'
   315→  else if (/id="impress"|impress\(\)\.init|\/frameworks\/impressjs\//i.test(html)) framework = 'impress.js'
   316→  else if (/class="shower"|\/frameworks\/shower\//i.test(html)) framework = 'shower'
   317→  else if (/aria-selected="true"|\/frameworks\/dzslides\//i.test(html) && /<section/i.test(html) && !/class="deck"/i.test(html)) framework = 'dzslides'
   318→  else framework = 'open-design'
   319→
   320→  // Extract just the design-relevant parts of the template HTML to keep the prompt tight.
   321→  const templateDigest = extractTemplateDigest(params.templateHtml, params.templateCss, framework)
   322→
   323→  // Pick the right deck structure contract based on framework
   324→  let deckContract: string
   325→  let notesTag: string
   326→  let inlineNote: string
   327→  switch (framework) {
   328→    case 'reveal.js':
   329→      deckContract = REVEAL_DECK_STRUCTURE
   330→      notesTag = '<aside class="notes">'
   331→      inlineNote = 'Keep <link>/<script> tags pointing to /revealjs/dist/... — do NOT inline reveal.js.'
   332→      break
   333→    case 'impress.js':
   334→      deckContract = IMPRESS_DECK_STRUCTURE
   335→      notesTag = '<div class="notes">'
   336→      inlineNote = 'Keep <link>/<script> tags pointing to /frameworks/impressjs/... — do NOT inline impress.js.'
   337→      break
   338→    case 'shower':
   339→      deckContract = SHOWER_DECK_STRUCTURE
   340→      notesTag = '<details class="notes">'
   341→      inlineNote = 'Keep <link>/<script> tags pointing to /frameworks/shower/... — do NOT inline shower core.'
   342→      break
   343→    case 'dzslides':
   344→      deckContract = DZSLIDES_DECK_STRUCTURE
   345→      notesTag = '<div role="note">'
   346→      inlineNote = 'Inline ALL CSS and JS — DZSlides is single-file, no external assets allowed.'
   347→      break
   348→    default:
   349→      deckContract = DECK_STRUCTURE
   350→      notesTag = '<div class="notes">'
   351→      inlineNote = 'Inline all CSS and JS.'
   352→  }
   353→
   354→  return `You are an expert presentation designer and frontend engineer. You generate single self-contained HTML presentation decks that look polished, professional, and on-brand.
   355→
   356→# IDENTITY
   357→- You write production-quality HTML/CSS/JS by hand. No frameworks. No build steps.
   358→- You think in design tokens (CSS variables), 4pt spacing scale, and Duarte-style narrative arcs.
   359→- You respect the chosen template's visual identity: fonts, palette, decorative motifs, and layout vocabulary are sacred.
   360→- You have access to a library of 57+ templates spanning 5 frameworks:
   361→  - 45 open-design HTML decks (custom hand-rolled CSS)
   362→  - 7 reveal.js themes (reveal.js 6.0.1 — comprehensive slide framework)
   363→  - 2 impress.js templates (3D spatial / Prezi-style camera transitions)
   364→  - 2 Shower themes (Ribbon + Material — modern modular)
   365→  - 1 DZSlides template (single-file minimalist, zero dependencies)
   366→- You detect the chosen template's framework — open-design, reveal.js, impress.js, shower, or dzslides — and follow that framework's deck structure contract exactly.
   367→
   368→# CAPABILITIES & CONSTRAINTS
   369→- Input: a topic brief from the user + a reference HTML template digest (the chosen or auto-selected template).
   370→- Output: ONE complete HTML file (the deck). Nothing else.
   371→- The chosen template is **${framework}** — you MUST follow the ${framework} deck structure contract below.
   372→- You may retune the template's :root tokens or --accent variable (e.g. shift accent hue) but you MUST preserve the template's font stack, class vocabulary, and structural primitives.
   373→- You MUST follow the deck structure contract, layout catalog, and strategies below.
   374→- If the chosen template's design feels mismatched to the content, still produce the deck honoring the template's visual identity — do not silently swap styles.
   375→
   376→# CONTEXT
   377→- Audience: ${audience}
   378→- Tone: ${tone}
   379→- Focus: ${focus}

   434→/**
   435→ * Extracts a digest of the template HTML to keep the AI prompt tight.
   436→ * Returns:
   437→ *   - The full <head> (fonts, tokens, base CSS, scoped CSS)
   438→ *   - The first 3 <section class="slide"> blocks (class vocabulary reference)
   439→ *   - The runtime <script> (keyboard nav)
   440→ *   - The optional style.css if present
   441→ */
   442→function extractTemplateDigest(html: string, css: string | undefined, framework: 'open-design' | 'reveal.js' | 'impress.js' | 'shower' | 'dzslides'): string {
   443→  const parts: string[] = []
   444→
   445→  // 1. Extract <head>...</head>
   446→  const headMatch = html.match(/<head[^>]*>([\s\S]*?)<\/head>/i)
   447→  if (headMatch) {
   448→    parts.push('=== <head> (fonts, tokens, base CSS) ===')
   449→    parts.push(headMatch[0])
   450→  }
   451→
   452→  // 2. Extract first 3 slide sections based on framework
   453→  let slideMatches: string[] | null = null
   454→  switch (framework) {
   455→    case 'reveal.js': {
   456→      const slidesBlock = html.match(/<div class="slides">([\s\S]*?)<\/div>\s*<\/div>/i)
   457→      if (slidesBlock) {
   458→        slideMatches = slidesBlock[1].match(/<section[^>]*>[\s\S]*?<\/section>/gi)
   459→      }
   460→      break
   461→    }
   462→    case 'impress.js': {
   463→      // <div class="step ..."> elements inside <div id="impress">
   464→      const impressBlock = html.match(/<div id="impress">([\s\S]*?)<\/div>\s*(?:<[^>]+>\s*)*<(?:script|\/body)/i)
   465→      if (impressBlock) {
   466→        slideMatches = impressBlock[1].match(/<div[^>]*class="[^"]*step[^"]*"[^>]*>[\s\S]*?<\/div>/gi)
   467→      }
   468→      break
   469→    }
   470→    case 'shower': {
   471→      // <section class="slide"> inside <body class="shower">
   472→      slideMatches = html.match(/<section[^>]*class="[^"]*slide[^"]*"[^>]*>[\s\S]*?<\/section>/gi)
   473→      break
   474→    }
   475→    case 'dzslides': {
   476→      // Plain <section> elements
   477→      slideMatches = html.match(/<section[^>]*>[\s\S]*?<\/section>/gi)
   478→      break
   479→    }
   480→    default: {
   481→      // open-design: <section class="...slide...">
   482→      slideMatches = html.match(/<section[^>]*class="[^"]*slide[^"]*"[^>]*>[\s\S]*?<\/section>/gi)
   483→    }
   484→  }
   485→  if (slideMatches && slideMatches.length > 0) {
   486→    parts.push('\n=== First slides (class vocabulary reference; do NOT copy content) ===')
   487→    parts.push(slideMatches.slice(0, 3).join('\n'))
   488→  }
   489→
   490→  // 3. Extract runtime <script> blocks
   491→  switch (framework) {
   492→    case 'reveal.js':
   493→    case 'impress.js':
   494→    case 'shower': {
   495→      // Include all <script> tags (both inline + src) — they reference the framework's runtime
   496→      const allScriptMatches = html.match(/<script[^>]*>[\s\S]*?<\/script>/gi)
   497→      if (allScriptMatches && allScriptMatches.length > 0) {
   498→        parts.push(`\n=== <script> tags (${framework} load + init — copy verbatim) ===`)
   499→        parts.push(allScriptMatches.join('\n'))
   500→      }
   501→      break
   502→    }
   503→    case 'dzslides': {
   504→      // Inlined JS only — copy verbatim
   505→      const scriptMatches = html.match(/<script(?![^>]*src=)[^>]*>[\s\S]*?<\/script>/gi)
   506→      if (scriptMatches && scriptMatches.length > 0) {
   507→        parts.push('\n=== Inlined <script> (keyboard nav — copy verbatim, no external src) ===')
   508→        parts.push(scriptMatches.join('\n'))
   509→      }
   510→      break
   511→    }
   512→    default: {
   513→      // open-design: only inline scripts (no src)
   514→      const scriptMatches = html.match(/<script(?![^>]*src=)[^>]*>[\s\S]*?<\/script>/gi)
   515→      if (scriptMatches && scriptMatches.length > 0) {
   516→        parts.push('\n=== Runtime <script> (keyboard nav — copy verbatim) ===')
   517→        parts.push(scriptMatches.join('\n'))
   518→      }
   519→    }
   520→  }
   521→
   522→  // 4. Append external style.css if present (open-design multi-file only)
   523→  if (css) {
   524→    parts.push('\n=== External style.css (scoped deck styles) ===')
   525→    parts.push(css)
   526→  }
   527→
   528→  return parts.join('\n')
   529→}
