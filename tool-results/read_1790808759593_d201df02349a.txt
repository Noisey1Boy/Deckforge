    37→function isTypingTarget(t: EventTarget | null): boolean {
    38→  const el = t as HTMLElement | null;
    39→  if (!el) return false;
    40→  const tag = el.tagName;
    41→  return tag === 'INPUT' || tag === 'TEXTAREA' || tag === 'SELECT' || el.isContentEditable;
    42→}

    44→function SlideCard({
    45→  slide, index, active, pageCount,
    46→  onSelect, onDuplicate, onDelete, onToggleHidden, onDragStart, onDragOver, onDrop,
    47→  dragging, dropTarget, selected,
    48→}: {
    49→  slide: Slide;
    50→  index: number;
    51→  active: boolean;
    52→  pageCount: number;
    53→  onSelect: (e: React.MouseEvent) => void;
    54→  onDuplicate: () => void;
    55→  onDelete: () => void;
    56→  onToggleHidden: () => void;
    57→  onDragStart: () => void;

    60→  dragging: boolean;
    61→  dropTarget: boolean;

    64→  const deck = useAppStore((s) => s.deck);
    65→  const themeManifest = deck ? deckTheme(deck) : getTheme(deck?.themeId ?? 'corporate-clean');
    66→  return (
    67→    <div
    68→      draggable
    69→      onDragStart={onDragStart}
    70→      onDragOver={onDragOver}
    71→      onDrop={onDrop}
    72→      className={cn(
    73→        'group relative rounded-lg overflow-hidden border-2 transition-all cursor-grab active:cursor-grabbing bg-card',
    74→        active ? 'shadow-md' : 'border-transparent hover:border-neutral-300',
    75→        dragging && 'opacity-40',
    76→        dropTarget && 'ring-2 ring-neutral-900/60',
    77→        selected && !active && 'ring-2 ring-emerald-500/70 ring-offset-1',
    78→        slide.hidden && !dragging && 'opacity-55 border-dashed'
    79→      )}
    80→      style={active ? { borderColor: themeManifest.tokens.colors.primary } : undefined}
    81→      onClick={onSelect}
    82→      role="button"
    83→      tabIndex={0}
    84→      onKeyDown={(e) => (e.key === 'Enter' || e.key === ' ') && onSelect(e as unknown as React.MouseEvent)}
    85→      aria-label={`Slide ${index + 1}: ${slide.title || slide.layout}${slide.hidden ? ' (hidden)' : ''}${selected ? ' (selected)' : ''}`}
    86→      aria-current={active}
    87→    >
    88→      <div className="pointer-events-none">
    89→        <ScaledSlide slide={slide} theme={themeManifest} index={index} pageCount={pageCount} showPageNumbers={false} />

    91→      <div className="absolute top-1 left-1 flex items-center gap-1">
    92→        <span className="bg-black/60 text-white text-[10px] rounded px-1.5 py-0.5 font-mono">{index + 1}</span>

    94→          {LAYOUT_ICONS[slide.layout]}

    97→          <span className="bg-neutral-900/80 text-white text-[9px] uppercase tracking-wider rounded px-1.5 py-0.5 font-semibold inline-flex items-center gap-0.5">
    98→            <EyeOff className="h-2.5 w-2.5" /> Hidden
    99→          </span>
   100→        )}
   101→      </div>
   102→      {selected && (
   103→        <span className="absolute bottom-1 right-1 bg-emerald-600 text-white rounded-full p-0.5 shadow" aria-hidden>
   104→          <Check className="h-3 w-3" />
   105→        </span>
   106→      )}
   107→      <div className="absolute top-1 right-1 hidden group-hover:flex gap-1">
   108→        <button
   109→          onClick={(e) => { e.stopPropagation(); onToggleHidden(); }}
   110→          className="bg-black/60 hover:bg-black/80 text-white rounded p-1"
   111→          aria-label={slide.hidden ? 'Show slide in presentation' : 'Hide slide from presentation'}
   112→          title={slide.hidden ? 'Show slide (H)' : 'Hide slide (H)'}
   113→        >
   114→          {slide.hidden ? <Eye className="h-3 w-3" /> : <EyeOff className="h-3 w-3" />}
   115→        </button>
   116→        <button
   117→          onClick={(e) => { e.stopPropagation(); onDuplicate(); }}

   120→        >

   124→          onClick={(e) => { e.stopPropagation(); onDelete(); }}

   126→          aria-label="Delete slide"

   128→          <Trash2 className="h-3 w-3" />
   129→        </button>
   130→      </div>
   131→    </div>
   132→  );
   133→}
