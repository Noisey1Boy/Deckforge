   110→export function CreateView() {
   111→  const setOutline = useAppStore((s) => s.setOutline);
   112→  const setView = useAppStore((s) => s.setView);
   113→  const setDataAssets = useAppStore((s) => s.setDataAssets);

   116→  const setDeck = useAppStore((s) => s.setDeck);
   117→  const reduceMotion = useReducedMotion();
   118→
   119→  const enter = (delay: number) =>
   120→    reduceMotion
   121→      ? {}
   122→      : {
   123→          initial: { opacity: 0, y: 18 },
   124→          animate: { opacity: 1, y: 0 },
   125→          transition: { duration: 0.55, delay, ease: [0.16, 1, 0.3, 1] as const },
   126→        };
   127→
   128→  const [prompt, setPrompt] = useState('');
   129→  const [audience, setAudience] = useState('');
   130→  const [tone, setTone] = useState<ToneId>('professional');
   131→  const [slideCount, setSlideCount] = useState(10);
   132→  const [language, setLanguage] = useState('English');
   133→  const [strategy, setStrategy] = useState<StrategyId | 'auto'>('auto');
   134→  const [verbosity, setVerbosity] = useState<Verbosity>('standard');
   135→  const [busy, setBusy] = useState<string | null>(null);
   136→  const [dataTab, setDataTab] = useState<'files' | 'image' | 'mermaid' | 'markdown'>('files');
   137→  // Start with a deterministic seed so SSR and the first client render match
   138→  // (avoids a hydration mismatch); deal a random set right after mount.
   139→  const [ideaSeed, setIdeaSeed] = useState(0);
   140→  useEffect(() => {
   141→    const t = window.setTimeout(() => setIdeaSeed(Math.floor(Math.random() * 3)), 0);
   142→    return () => window.clearTimeout(t);
   143→  }, []);
   144→

   146→  const [tables, setTables] = useState<ParsedTable[]>([]);
   147→  const [chartSpecs, setChartSpecs] = useState<ChartSpec[]>([]);
   148→  const [diagram, setDiagram] = useState<MermaidPlan | null>(null);
   149→  const [mermaid, setMermaid] = useState('');
   150→  const [markdown, setMarkdown] = useState('');
   151→  const [dataError, setDataError] = useState<string | null>(null);
   152→  const fileRef = useRef<HTMLInputElement>(null);
   153→  const imgRef = useRef<HTMLInputElement>(null);
   154→
   155→  // ---------------------------------------------------------------- data
   156→  const handleFiles = async (files: FileList | null) => {
   157→    if (!files?.length) return;
   158→    setDataError(null);
   159→    setBusy('Reading data…');
   160→    const newTables: ParsedTable[] = [...tables];
   161→    for (const file of Array.from(files).slice(0, 3)) {
   162→      const { tables: ts, error } = await parseFileToTables(file);
   163→      if (error) {
   164→        setDataError(error);
   165→        toast.error(error);
   166→      }
   167→      newTables.push(...ts);
   168→    }
   169→    setTables(newTables.slice(0, 3));
   170→    setChartSpecs([]); // re-plan after new data
   171→    setBusy(null);
   172→  };
   173→
   174→  const handleChartImage = async (files: FileList | null) => {
   175→    const file = files?.[0];
   176→    if (!file) return;
   177→    if (!/^image\/(png|jpe?g|webp)$/.test(file.type)) {
   178→      toast.error('Use a PNG, JPG or WebP screenshot of your chart.');
   179→      return;
   180→    }
   181→    if (file.size > 6_000_000) {
   182→      toast.error('Image must be under 6 MB.');
   183→      return;
   184→    }
   185→    setBusy('Reading chart from image…');
   186→    try {
   187→      const dataUrl = await new Promise<string>((resolve, reject) => {
   188→        const r = new FileReader();
   189→        r.onload = () => resolve(String(r.result));
   190→        r.onerror = reject;
   191→        r.readAsDataURL(file);
   192→      });
   193→      const res = await fetch('/api/data/chart-image', {
   194→        method: 'POST',
   195→        headers: { 'Content-Type': 'application/json' },
   196→        body: JSON.stringify({ image: dataUrl, language }),
   197→      });
   198→      const json = await res.json();
   199→      if (!res.ok) throw new Error(json.error || 'Extraction failed');
   200→      setChartSpecs((prev) => [...prev, json.chart as ChartSpec]);
   201→      toast.success('Chart extracted from image.');
   202→    } catch (e) {
   203→      toast.error(e instanceof Error ? e.message : 'Image analysis failed');
   204→    } finally {
   205→      setBusy(null);
   206→      if (imgRef.current) imgRef.current.value = '';
   207→    }
   208→  };
   209→

   211→    if (mermaid.trim().length < 10) {
   212→      toast.error('Paste a Mermaid diagram first (e.g. "graph LR; A-->B;").');
   213→      return;
   214→    }
   215→    setBusy('Converting diagram…');
   216→    try {
   217→      const res = await fetch('/api/data/analyze', {
   218→        method: 'POST',
   219→        headers: { 'Content-Type': 'application/json' },
   220→        body: JSON.stringify({ mermaid, language }),
   221→      });
   222→      const json = await res.json();
   223→      if (!res.ok) throw new Error(json.error || 'Conversion failed');
   224→      if (json.diagram) {
   225→        setDiagram(json.diagram as MermaidPlan);
   226→        toast.success('Diagram ready to include.');
   227→      } else throw new Error('Could not parse the diagram');
   228→    } catch (e) {
   229→      toast.error(e instanceof Error ? e.message : 'Conversion failed');
   230→    } finally {
   231→      setBusy(null);
   232→    }
   233→  };
   234→
   235→  // ------------------------------------------------------------- generate
   236→  const generateOutline = async () => {
   237→    if (prompt.trim().length < 3) {
   238→      toast.error('Describe what the presentation should be about.');
   239→      return;
   240→    }
   241→    setBusy('Designing the outline…');
   242→    try {
   243→      // Persist generation controls so the slide pipeline uses the same settings.
   244→      const chosen = { audience, tone, language, slideCount, strategy, verbosity };
   245→      setGenSettings(chosen);
   246→      let charts: ChartSpec[] = [...chartSpecs];
   247→      let diagramPlan: MermaidPlan | undefined = diagram ?? undefined;
   248→      let dataContext: string | undefined;
   249→
   250→      const wantsAnalysis = tables.length > 0;
   251→      if (wantsAnalysis) {
   252→        setBusy('Analyzing your data…');
   253→        const res = await fetch('/api/data/analyze', {
   254→          method: 'POST',
   255→          headers: { 'Content-Type': 'application/json' },
   256→          body: JSON.stringify({ tables: tables.map(tableToMeta), language }),
   257→        });
   258→        const json = await res.json();
   259→        if (!res.ok) throw new Error(json.error || 'Analysis failed');
   260→        const plans = (json.charts ?? []) as ChartPlan[];
   261→        const specs = plans
   262→          .map((p) => {
   263→            const t = tables[p.tableIndex];
   264→            return t ? buildChartSpec(t, p) : null;
   265→          })
   266→          .filter((c): c is ChartSpec => !!c);
   267→        charts = [...charts, ...specs];
   268→        dataContext = json.dataContext;
   269→      }
   270→      if (diagramPlan) {
   271→        dataContext = `${dataContext ?? ''}\nA Mermaid flow diagram "${diagramPlan.title}" with ${diagramPlan.nodes.length} steps is available for a "diagram" slide.`;
   272→      }
   273→
   274→      setBusy('Writing the outline…');
   275→      const res = await fetch('/api/generate/outline', {
   276→        method: 'POST',
   277→        headers: { 'Content-Type': 'application/json' },
   278→        body: JSON.stringify({
   279→          prompt, audience, tone, slideCount, language, themeChoice, dataContext, strategy, verbosity,
   280→        }),
   281→      });
   282→      const json = await res.json();
   283→      if (!res.ok) throw new Error(json.error || 'Outline failed');
   284→      const outline = json.outline as DeckOutline;
   285→
   286→      // When the strategy was Auto, remember the resolved one so slide
   287→      // generation (and the outline UI) names the actual narrative used.
   288→      if (strategy === 'auto' && typeof json.strategyUsed === 'string') {
   289→        setGenSettings({ ...chosen, strategy: json.strategyUsed as StrategyId });
   290→      }
   291→
   292→      // attach data: first N chart slides in the outline get real chart specs
   293→      const chartAssignments: Record<string, ChartSpec> = {};
   294→      const pool = [...charts];
   295→      let remainingDiagram = diagramPlan;
   296→      outline.slides.forEach((s, i) => {
   297→        if (s.layout === 'chart' && pool.length) {
   298→          chartAssignments[String(i)] = pool.shift()!;
   299→        }
   300→      });
   301→      // if diagram exists but no diagram slide planned, convert the last chart-less content slide? Keep: AI was told about it.
   302→
   303→      setDataAssets({
   304→        tables,
   305→        charts,
   306→        diagram: remainingDiagram ? { ...remainingDiagram } : undefined,
   307→        dataContext,
   308→        chartAssignments,
   309→      });
   310→      setOutline(outline);
   311→      setView('outline');
   312→    } catch (e) {
   313→      toast.error(e instanceof Error ? e.message : 'Generation failed');
   314→    } finally {
   315→      setBusy(null);
   316→    }
   317→  };
   318→
   319→  // ------------------------------------------------- markdown → outline
   320→  const buildFromMarkdown = () => {
   321→    if (markdown.trim().length < 10) {
   322→      toast.error('Paste some markdown first; headings become slides.');
   323→      return;
   324→    }
   325→    try {
   326→      const parsed = parseMarkdownOutline(markdown);
   327→      if (parsed.slides.length < 1) throw new Error('No headings found');
   328→      // persist generation controls so the slide pipeline uses the same settings
   329→      setGenSettings({ audience, tone, language, slideCount, strategy, verbosity });
   330→      const outline: DeckOutline = {
   331→        title: parsed.title,
   332→        subtitle: parsed.subtitle || prompt.trim().slice(0, 110),
   333→        slides: parsed.slides,
   334→      };
   335→      // deterministic theme suggestion when Auto is selected
   336→      if (themeChoice === 'auto') {
   337→        outline.recommendedThemeId = suggestThemeId(markdown);
   338→        outline.themeRationale = 'Matched to your markdown content.';
   339→      }
   340→      // attach user charts to slides tagged [chart], in order
   341→      const chartAssignments: Record<string, ChartSpec> = {};
   342→      const pool = [...chartSpecs];
   343→      outline.slides.forEach((s: OutlineItem, i: number) => {
   344→        if (s.layout === 'chart' && pool.length) {
   345→          chartAssignments[String(i)] = pool.shift()!;
   346→        }
   347→      });
   348→      setDataAssets({
   349→        tables, charts: chartSpecs,
   350→        diagram: diagram ?? undefined,
   351→        dataContext: undefined,
   352→        chartAssignments,
   353→      });
   354→      setOutline(outline);
   355→      setView('outline');
   356→      toast.success(`Outline built — ${outline.slides.length} slides from your markdown.`);

   358→      toast.error(e instanceof Error ? e.message : 'Could not parse markdown');
   359→    }

   378→
   379→  // ---- drop a data file anywhere on the page to attach it ---------------

   389→  };

   391→    if (!e.dataTransfer.types.includes('Files')) return;

   405→      const dt = new DataTransfer();
   406→      images.forEach((f) => dt.items.add(f));
   407→      void handleChartImage(dt.files);
   408→    }
   409→  };
   410→
   411→  return (
   412→    <div className="w-full" onDragOver={pageDrag} onDragLeave={pageDragLeave} onDrop={pageDrop}>
   413→      {droppingData && (
   414→        <div className="fixed inset-2 z-40 grid place-items-center border-2 border-dashed border-[color:var(--stg-yellow)] bg-[color:var(--stg-ink)]/95 pointer-events-none" aria-hidden>
   415→          <div className="text-center">
   416→            <FileUp className="h-8 w-8 mx-auto text-[color:var(--stg-yellow)]" />
   417→            <p className="stg-display mt-2 text-2xl uppercase text-[color:var(--stg-cream)]">Drop your data</p>
   418→            <p className="stg-label mt-1 text-[color:var(--stg-gray)]">CSV · XLSX · JSON AS DATA · IMAGES AS CHART SCREENSHOTS</p>

   420→        </div>

   422→
   423→      {/* ================================================================
   424→          HERO — the stage. Ink, poster yellow, and the screen.

   426→      <section className="relative overflow-hidden bg-[color:var(--stg-ink)] text-[color:var(--stg-cream)]">
   427→        <div className="relative mx-auto max-w-7xl px-4 sm:px-6 pt-16 sm:pt-24 pb-14 sm:pb-20">
   428→          <div className="grid items-center gap-12 lg:grid-cols-12 lg:gap-14">
   429→            {/* Copy */}
   430→            <motion.div {...enter(0)} className="lg:col-span-6 space-y-7">
   431→              <p className="stg-label flex items-center gap-3 text-[color:var(--stg-yellow)]">
   432→                <span className="inline-block h-2 w-2 bg-[color:var(--stg-yellow)]" aria-hidden />
   433→                DeckForge AI · A presentation foundry
   434→              </p>
   435→              <h1 className="stg-display text-[clamp(3.1rem,7vw,5.9rem)] uppercase leading-[0.93]">
   436→                <span className="block">The lights go down.</span>
   437→                <span className="block text-[color:var(--stg-yellow)]">Your story holds the room.</span>
   438→              </h1>
   439→              <p className="max-w-[62ch] text-base sm:text-lg leading-relaxed text-[color:var(--stg-gray)]">
   440→                Give DeckForge one sentence, or hand it your spreadsheet. It drafts the
   441→                outline, you approve the story, and every slide is designed against a real
   442→                design system with charts built from your actual numbers.
   443→              </p>
   444→              <div className="flex flex-wrap items-center gap-3 pt-1">
   445→                <a

   447→                  className="stg-label inline-flex h-12 items-center gap-2 bg-[color:var(--stg-yellow)] px-6 text-[color:var(--stg-ink)] transition-colors hover:bg-[color:var(--stg-yellow-deep)] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[color:var(--stg-yellow)] focus-visible:ring-offset-2 focus-visible:ring-offset-[color:var(--stg-ink)]"
